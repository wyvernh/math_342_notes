\section{First Order Vector-Valued ODE}\label{sec:first-order}

\subsection{Repeated eigenvalues}

\begin{enumerate}[itemsep=10pt]
\item Example: Find a fundamental set for
  $\dot x = \begin{bmatrix} 1 & -1 \\ 4 & 5\end{bmatrix}x$.
  \begin{align}
    \lambda &= 3,\quad \xi = [1,-2]^T\\
    (d/dt-\lambda)^2 x &= 0 \text{ has solns } Ce^{\lambda t},\, Cte^{\lambda t}\\
  \end{align}
  Propose
  \begin{align}
    x = te^{\lambda t}\xi\,?
  \end{align}
  but then
  \begin{align}
    \dot x - Ax = e^{\lambda t}\xi \neq 0
  \end{align}
  so $x$ is not a solution.

  However, we can cancel off the nonhomogeneous part with an ansatz solution $e^{\lambda t}\eta$, $\eta \in \RR^2$, to the nonhomogenous equation $\dot x - Ax = e^{\lambda t}\xi$. (Compare with method of undetermined coefficients.)

  So
  \begin{align}
    x = te^{\lambda t}\xi - e^{\lambda t}\eta.
  \end{align}

  Then $\dot x - Ax = 0$ gives
  \begin{align}
    0 = e^{\lambda t}\xi + (\lambda e^{\lambda t}\eta - Ae^{\lambda t}\eta),
  \end{align}
  and therefore
  \begin{align}
    e^{\lambda t}(\xi + \lambda\eta - A\eta) = 0
    &\implies \xi + \lambda\eta - A\eta = 0\\
    &\implies \xi = (A - \lambda I)\eta.
  \end{align}
  Notice that $A - \lambda I$ is not invertible. But concretely we want to solve
  \begin{align}
    \begin{bmatrix}
      -2 & -1\\
      4 & 2
    \end{bmatrix}
    \begin{bmatrix}
      \eta_1 \\ \eta_2
    \end{bmatrix}
    =
    \begin{bmatrix}
      1 \\ -2
    \end{bmatrix}.
  \end{align}
  Is $\xi$ always in the column space of $A - \lambda I$?

\item Summary (choice of a fundamental set).

  \emph{In general} when $A$ has repeated eigenvalues, and $A$ is $2\times 2$,
  \begin{align}
    x^1 &= e^{\lambda t}\xi,\quad
    x^2 = te^{\lambda t}\xi + e^{\lambda t}\eta,
  \end{align}
  where $\eta$ is a solution to $(A-\lambda I)\eta = \xi$.

  $\eta$ is called a \emph{generalised eigenvector}.

  \emph{Highly important result}. If $\eta$ is \emph{any} vector independent of $\xi$, we will have $(A - \lambda I)\eta = c\xi$ for some $c \neq 0$. (Why?)

  \end{enumerate}

\subsection{Nonhomogeneous system}
\begin{enumerate}
  \item The system of interest is
  \begin{equation}\label{eq:nonhomogeneous-system}
    \dot x = P(t)x + g(t),\quad\quad P(t) \in M_{n\times n}(\RR),\quad x(t),g(t) \in M_{n\times 1}(\RR)
  \end{equation}
  Solution is given as a sum of a particular solution with a general solution of the corresponding homogeneous system (which we have already done). We will focus on finding particular solutions $x_p$ to (\ref{eq:nonhomogeneous-system}).

\item Methods:
  \begin{enumerate}
  \item Undetermined coefficients for special $g(t)$
  \item Laplace transforms
  \item Decoupled systems for diagonalizable $A$
  \item \textbf{Variation of parameters} (we will use this one)
  \end{enumerate}

  \underline{Example:}
  $\dot x = \begin{bmatrix} 1 & -1 \\ 4 & 5\end{bmatrix}x + \begin{bmatrix} e^t \\ 0 \end{bmatrix}$.

\item
  Proposed solutions (VOP):
  \begin{equation}
    x(t) = u_1(t)x^1(t) + u_2(t)x^2(t) \quad\quad\text{(undetermined coefficients)}
  \end{equation}
  where $u_1,u_2$ are scalar valued and $x^1$,$x^2$ solve the homogeneous equation. The goal is to find $u_1,u_2$.
  \begin{enumerate}
  \item Rewrite
    \begin{align}
      x(t) &=
      \begin{bmatrix}
        x^1(t) & x^2(t)
      \end{bmatrix}
      \begin{bmatrix}
        u_1(t) \\ u_2(t)
      \end{bmatrix}\\
      &= \Psi(t)u(t)\\
      &=
      \begin{bmatrix}
        e^t & e^{5t}\\
        -3e^t & e^{5t}
      \end{bmatrix}
      \begin{bmatrix}
        u_1(t) \\ u_2(t)
      \end{bmatrix}
    \end{align}

  \item Apply $x(t) = \Psi(t)u(t)$ to $\dot x(t) = Ax(t) + g(t)$ to get
    \begin{equation}\label{eq:cond_on_VOP_coeffs}
      \Psi'(t)u(t) + \Psi(t)u'(t) = A\Psi(t)u(t) + g(t)
    \end{equation}

  \item Remember $\Psi(t)$ is invertible, and also explicitly
    \begin{align}
      \Psi(t) &=
      \begin{bmatrix}
        x^1(t) & x^2(t)
      \end{bmatrix}\\
      \Psi'(t) &=
      \begin{bmatrix}
        Ax^1(t) & Ax^2(t)
      \end{bmatrix}\\
      &= A\Psi(t).
    \end{align}
    $\Psi$ solves a ``matrix version'' of the original homogeneous equation. We use this result to simplify (\ref{eq:cond_on_VOP_coeffs}):
    \begin{align}
      \Psi(t)u'(t) = g(t).
    \end{align}

  \item We have
    \begin{align}
      \begin{bmatrix}
        e^t & e^{5t}\\
        -3e^t & e^{5t}
      \end{bmatrix}
      \begin{bmatrix}
        u_1'(t) & u_2'(t)
      \end{bmatrix}
      =
      \begin{bmatrix}
        e^t & 0
      \end{bmatrix},
    \end{align}
    so
    \begin{align}
      \begin{cases}
        e^tu_1'(t) + e^{5t}u_2'(t) = e^t\\
        -3e^tu_1'(t) + e^{5t}u_2'(t) = 0
      \end{cases}
    \end{align}
    A solution is
    \begin{align}
      \begin{cases}
        u_1'(t) = 1/4\\
        u_2'(t) = \frac{3}{4}e^{-4t}
      \end{cases}
    \end{align}

  \item Recover $u(t)$ from $u'(t)$.
    \begin{align}
      u(t) = \int u'(t)dt =
      \int
      \begin{bmatrix}
        \frac{1}{4} \\ \frac{3}{4}e^{-4t}
      \end{bmatrix}
      dt
      =
      \begin{bmatrix}
        \frac{1}{4}t \\ -\frac{3}{16}e^{-4t}
      \end{bmatrix}
      + c
    \end{align}

  \item One particular solution is
    \begin{align}
      x(t) = \left(\int_0^t\frac{1}{4}ds\right)
      \begin{bmatrix}
        e^t \\ -3e^t
      \end{bmatrix}
      +
      \left(\int_0^t\frac{3}{4}e^{-4s}ds\right)
      \begin{bmatrix}
        e^{5t} \\ e^{5t}
      \end{bmatrix}.
    \end{align}

  \end{enumerate}

\item \emph{Solution method in general.}

  \underline{Problem} $$\dot x(t) = P(t)x(t) + g(t).$$

  \underline{Corresponding homogeneous system} $$\dot x(t) = P(t)x(t),$$ solved by $$\dot x_h = P(t)x_h.$$

  \underline{Solution to the nonhomogeneous system} $$x(t) = \Psi(t)u(t)$$ (solve for $u(t)$).

  \underline{Equation for $u(t)$} $$\Psi'(t)u(t) + \Psi(t)u'(t) = P(t)\Psi(t)u(t) + g(t).$$
  OBSERVE that $\Psi'(t) = P(t)\Psi(t)$ (even in the nonconstant $P(t)$ case), so that
  $$\Psi(t)u(t) = g(t).$$
  Furthermore, \emph{$\Psi$ is invertible.} This is because of the whole Wronskian nonzero $\implies$ nonzero everywhere / diffeq's do not collapse the dimension of subspaces spanned by solutions thing.
  Thus
  \begin{align}
    u(t) &= \int \Psi(t)^{-1}g(t)dt\\
    \implies x_p(t) &= \Psi(t)\left(c + \int_0^t \Psi(s)^{-1}g(s)ds\right)\\
    &= \Psi(t)c + \int_0^t\Psi(t)\Psi(s)^{-1}g(s)ds,
  \end{align}
  (which is even the general solution to the original equation, with degree of freedom $c$ to be determined by initial conditions).

  \textbf{Special case:} $P(t) = A$ (constant). Then
  \begin{align}
    x(t) = e^{tA}x(0) = \int_0^te^{(t-s)A}g(s)ds \quad\quad\text{(Duhamel's formula)}
  \end{align}

\end{enumerate}

\section{Power Series Solutions to ODE}

\subsection{Basic power series tools}
\begin{enumerate}
\item Example.
  \begin{equation}\label{eq:power-series-function-ex}
    f(x) = \sum_{n=1}^\infty a_nx^n = \sum_{n=1}^\infty\frac{2^n}{n}x^n.
  \end{equation}
  \begin{enumerate}
  \item Find the radius of convergence.

    The radius of convergence of a power series $f(x) = \sum_{n=1}^\infty a_n(x-x_0)^n$ is a number $R \geq 0$ such that
    \begin{enumerate}
    \item $\forall x$ satisfying $|x - x_0| < R$, $f(x)$ converges absolutely (as in, $\sum_{n=1}^\infty |a_n||x-x_0|^n$ converges).
      \item $\sum_{n=1}^\infty a_n(x-x_0)^n$ diverges whenever $|x - x_0| > R$.
    \end{enumerate}
    $R$ is called the \emph{radius of convergence}.

    (Here $R = 1/2$ by the Ratio Test.)\\

    \underline{Tools for finding the radius of convergence}

    \textbf{The Root Test:} If $\ds\lim_{n\to\infty}|a_n|^{1/n}$ converges, then $\ds R = \frac{1}{\lim_{n\to\infty}|a_n|^{1/n}}$.

    For instance, (\ref{eq:power-series-function-ex}) gives
    \begin{equation}
      R = \frac{1}{\lim_{n\to\infty}(2^n/n)^{1/n}} = \frac{1}{\lim_{n\to\infty}2/n^{1/n}}= 1/2\quad\quad\text{since $n^{1/n}\to 1$}
    \end{equation}

    \textbf{The Ratio Test:} If $\ds\lim_{n\to\infty}\frac{|a_{n+1}|}{|a_n|}$ converges, then $\ds R = \frac{1}{\lim_{n\to\infty}\frac{|a_{n+1}|}{|a_n|}}$.

    In our example,
    \begin{equation}
      R = \frac{1}{\lim_{n\to\infty}\frac{2^{n+1}/(n+1)}{2^n/n}} = \frac{1}{\lim_{n\to\infty}2n/n+1}= 1/2.
    \end{equation}

    \bigskip

  \item Find $f'(x)$ (for meaningful $x$)

    As usual, differentiate term by term. If $f(x)$ is given by a power series $\sum_{n=0}^\infty a_nx^n$, then
    \begin{equation}\label{eq:power-series-function-derivative}
      f'(x) = \sum_{n=0}^\infty na_nx^{n-1} = \sum_{n=1}^\infty na_nx^{n-1}
    \end{equation}
    By making a change of variable $k = n-1$, and noticing that the first term drops out of the above series (because $n=0$), we have
    \begin{equation}\label{eq:derivative-power-series}
      f'(x) = \sum_{k=0}^\infty (k+1)a_{k+1}x^k.
    \end{equation}
    This procedure for getting the power series into the desired powers of $x$ is called \emph{index shifting}. It is important to keep track of the initial value for $k$ and $n$ in the sum. This is analagous to changing the bounds of integration when performing a change of variable.

    A remarkable feature of the radius of convergence is that it is \emph{invariant under differentiation or integration of the power series}. Consider our example. Differentiating term by term or using (\ref{eq:derivative-power-series}), the derivative series is
    \begin{equation}
      f'(x) = \sum_{k=0}^\infty (k+1)\frac{2^{k+1}}{k+1}x^k
      = \sum_{n=0}^\infty 2^{n+1}x^n.
    \end{equation}
    By the ratio test, the radius of convergence of $f'(x)$ is
    \begin{equation}
      R = \frac{1}{\lim_{n\to\infty}\frac{2^{n+2}}{2^{n+1}}}
      = \frac{1}{2},
    \end{equation}
    which is the same as the radius of convergence for $f(x)$. The radius of convergence of the integral power series $\int dx f(x)$ is a fun exercise (it should also be $1/2$).

  \item \underline{Some notions.} Let
    \begin{equation}
      f(x) = \sum_{n=0}^\infty a_n(x-x_0)^n \quad\quad\text{with}\quad\quad R>0.
    \end{equation}
    \begin{enumerate}
    \item $f(x)$ is \emph{analytic} at $x = x_0$. (This is what it means to be analytic at $x_0$; there exists a convergent power series equal to the function on a neighborhood of $x_0$.)
    \item $a_n = f^{(n)}(x_0)/n!$, and $f(x) = a_0 + a_1(x - x_0) + a_2(x - x_0)^2 + \dots$, and in particular $f(x_0) = a_0$.
    \item Taylor series expansion: When $f$ is analytic at $x_0$, it has a particular power series representation for $f(x)$, given by
      \begin{equation}
        f(x) = \sum_{n=0}^\infty \frac{f^{(n)}(x_0)}{n!}(x-x_0)^n
      \end{equation}
      For example, the exponential function is its own derivative and $e^0 = 1$. This is enough to define the exponential everywhere by its Taylor series at $x_0 = 0$:
      \begin{equation}
        e^x = \sum_{n=0}^\infty \frac{(e^x)^{(n)}\big|_{x=0}}{n!}x^n = \sum_{n=0}^\infty \frac{e^0}{n!}x^n = \sum_{n=0}^\infty \frac{1}{n!}x^n,
      \end{equation}
      which should look familiar.

    \end{enumerate}

  \end{enumerate}

\item Analytic vs infinitely differentiable functions.

  \underline{Question:} Every function that is analytic at $x=0$ is infinitely differentiable at $x=0$. Is the converse true? NO!

  \underline{Example:}
  A classic example is the function $f:\RR \to \RR$ given by
  \begin{equation}
    f(x) =
    \begin{cases}
      e^{-1/x^2}, & x > 0\\
      0, & x \leq 0
    \end{cases}
  \end{equation}
  This function is infinitely differentiable at $x_0 = 0$ (why?) but is not analytic there. For if it were analytic, its Taylor series would be identically zero since $f^{(n)}(0) = 0$ for all $n$ (look at the value of $f$ for negative $x$ arbitrarily close to zero).

  \begin{center}
    \includegraphics[width=0.8\textwidth]{nonanalytic_function}
  \end{center}

  \underline{Fact:} $\sum_{n=0}^\infty a_nx^n = 0 \implies a_n = 0$ for all $n$. This is just a restatement of what we said above. (If you define $f(x)$ by that power series then we identify the $a_n$ with the derivatives of $f$, which are all zero since $f$ is by assumption identically zero.)

\end{enumerate}

\subsection{Homogeneous DE with variable coefficients}

Consider the general type of ordinary differential equation
\begin{equation}\label{eq:homogeneous-de-with-variable-coeff}
  P(x)y'' + Q(x)y' + R(x)y = 0\quad\quad (y = y(x) \text{ is real valued})
\end{equation}
We will make the following assumption.

\underline{Assumption:} $P,Q,R$ are polynomials and there is no scalar $c$ such that $P(x),Q(x),R(x)$ are all divisible by $(x-c)$ (they are essentially ``coprime'' as elements of the ring of polynomials over the real or complex numbers).

\begin{enumerate}
\item Well known equations.
  \begin{enumerate}
  \item Airy's equation:
    \begin{equation}
      y'' - xy = 0,\quad\quad P(x) = 1, Q = 0, R(x) = -x.
    \end{equation}
  \item Bessel's equation:
    \begin{equation}
      x^2y'' + xy' + (x^2 - \nu^2)y = 0,\quad\quad P(x) = x^2, Q(x) = x, R(x) = x^2 - \nu^2
    \end{equation}
    where $\nu$ (the Greek letter ``nu'') is a parameter in the equation.

    If $\nu \neq 0$, then our assumption (about the coprime nature of $P,Q,R$) is satisfied, but not if $\nu = 0$. In the case $\nu = 0$, the equation simplifies a fair bit.

  \item Legendre's equation:
    \begin{equation}
      (1 - x^2)y'' - 2xy' + \alpha(\alpha + 1)y = 0,
    \end{equation}
    where $\alpha$ is a parameter.

  \end{enumerate}

\item Solution methods.

  Let us restate our DE in \textbf{standard form}.
  \begin{equation}
    y'' + \frac{Q}{P}(x)y' + \frac{R}{P}(x)y = 0
  \end{equation}
  or
  \begin{equation}
    y'' + p(x)y' + q(x)y = 0,
  \end{equation}
  where $p(x) = \frac{Q}{P}(x)$, and $q(x) = \frac{R}{P}(x)$.

  But wait! We need to verify that it is okay to divide by $P(x)$ (what if we are dividing by zero?). If $P(x_0) \neq 0$, then $P(x) \neq 0$ for all $x$ near $x_0$, and the standard form is valid near $x_0$. In this situation, $x_0$ is called an \textbf{ordinary point}.

\item \underline{Example:} Airy's equation.
  \begin{equation}\label{eq:airys-eq-example}
    y'' - xy = 0 \quad\quad (x\in\RR)
  \end{equation}
  We can do some quantitative analysis on Airy's equation and make some rough sketches of various solution curves to (\ref{eq:airys-eq-example}). This essentially boils down to using the equation to guess the sign and magnitude of $y''$ at different points on the sketch, and deducing the shape of the solution by forcing it to be consistent with this data. Unfortunately I can't draw in \LaTeX, so refer to the course notes for details.

  To proceed with the solution we use a power series ansatz
  \begin{equation}
    y(x) = \sum_{n=0}^\infty a_n(x-x_0)^n = \sum_{n=0}^\infty a_nx^n
  \end{equation}
  where $x_0 = 0$.

  Solving the (\ref{eq:airys-eq-example}) amounts to determining the coefficients $a_n$. To do this we need to plug $y(x)$ into (\ref{eq:airys-eq-example}). Let's calculate the derivatives (see the short section on power series):
  \begin{align}
    y'(x) &= \sum_{n=0}^\infty \frac{d}{dx}\left[a_nx^n\right]\\
    &= \sum_{n=1}^\infty na_nx^{n-1}\\
    y''(x) &= \sum_{n=1}^\infty \frac{d}{dx}\left[na_nx^{n-1}\right]\\
    &= \sum_{n=2}^\infty n(n-1)a_nx^{n-2}.
  \end{align}
  The lower limit of the sums is \emph{very important}. We drop terms when they are zero so that we don't end up with $x$ to negative powers in the derivative (e.g. $na_n = 0$ when $n=0$).

  Plugging into the differential equation, we have
  \begin{align}
    \sum_{n=2}^\infty n(n-1)a_nx^{n-2} - x\sum_{n=0}^\infty a_nx^n &= 0\\
    \implies\sum_{n=2}^\infty n(n-1)a_nx^{n-2} - \sum_{n=0}^\infty a_nx^{n+1} &= 0
  \end{align}
  Doing some index shifting with $k = n-2$ in the first sum and $k = n+1$ in the second sum, we get
  \begin{align}\label{eq:airys-power-series}
    \sum_{k=0}^\infty (k+2)(k+1)a_{k+2}x^k - \sum_{k=1}^\infty a_{k-1}x^k &= 0.
  \end{align}
  Writing out the first few terms, you can convince yourself that we have
  \begin{align}\label{eq:expansion-of-airys-equation}
    (2a_2 + 6a_3x + 12a_4x^2 + 20a_5x^3 + \dots) - (a_0x + a_1x^2 + a_2x^3 + a_3x^4 + \dots) &= 0.
  \end{align}
  Now we can simply collect terms with the same powers of $x$ and require that all of the coefficients are zero. We can do this because $1$, $x$, $x^2$, $x^3$ $\dots$ are linearly independent functions on an \emph{interval} (they are naturally called the Taylor basis for polynomials and power series). Since we know that the resulting power series is zero for $x$ in an interval (this technically comes from our hypothesis that the ansatz $y(x)$ is a solution), we can use this to determine $a_n$ for all $n$.

  We could use (\ref{eq:expansion-of-airys-equation}) to determine $a_0,a_1,a_2,\dots$, but we would never find them all. So let's do it formally. From (\ref{eq:airys-power-series}), we have
  \begin{equation}
    2a_2 + \sum_{k=1}^\infty \left[a_{k+2}(k+2)(k+1) - a_{k-1}\right]x^k = 0,
  \end{equation}
  which gives us the equations
  \begin{equation}
    \begin{cases}
      2a_2 = 0,\\
      a_{k+2}(k+2)(k+1) - a_{k-1} = 0\quad\forall k\geq 1
    \end{cases}
  \end{equation}
  The first equation gives $a_2 = 0$, and the second can be arranged to be
  \begin{align}
    a_{k+2} &=  - \frac{a_{k-1}}{(k+2)(k+1)}\quad\forall k\geq 1\\
    \implies a_{k+3} &=  - \frac{a_k}{(k+3)(k+2)}\quad\forall k\geq 1.
  \end{align}
  So any coefficient determines the coefficient three terms later in the series $f(x)$. Once we have $a_0$, we can generate $a_3$, and then $a_6$, and so on. Naturally, the coefficients form three decoupled sequencies or ``chains''
  \begin{align}
    &a_0,a_3,a_6,a_9,\dots,a_{3n},\dots\\
    &a_1,a_4,a_7,a_{10},\dots,a_{3n+1},\dots\\
    &a_2,a_5,a_8,a_{11},\dots,a_{3n+2},\dots = 0
  \end{align}
  The last of these is constant zero because $a_2 = 0$. The other two are controlled by two free parameters $a_0$ and $a_1$. By computing a few of these $a_n$s you will quickly realise a pattern, which you can write down as an explicit formula for coefficients in the first two chains:
  \begin{align}
    a_{3n} &= \frac{a_0}{\prod_{k=1}^n(3k-1)(3k)}\\
    a_{3n+1} &= \frac{a_1}{\prod_{k=1}^n3k(3k+1)}
  \end{align}
  (recall the notation $\prod_{k=1}^n q_k = q_1\cdot q_2\cdots q_n$).

  With this information we can simplify our ansatz and obtain the general solution.
  \begin{align}
    y(x) &= \sum_{n=0}^\infty a_nx^{n}\\
    &= \sum_{n=0}^\infty a_{3n}x^{3n} + \sum_{n=0}^\infty a_{3n+1}x^{3n+1}\\
    &= \sum_{n=0}^\infty \frac{a_0}{\prod_{k=1}^n(3k-1)(3k)}x^{3n} + \sum_{n=0}^\infty \frac{a_1}{\prod_{k=1}^n3k(3k+1)}x^{3n+1}\\
    &= a_0y_1(x) + a_1y_2(x),
  \end{align}
  where $y_1(x)$ and $y_2(x)$ are the two linearly independent solutions given by the power series in the previous line. This completes the derivation of the general solution to Airy's equation.

  \underline{Radius of convergence}
  Remember that power series functions are only defined within their radius of convergence. Let's check where our solution to the Airy's equation is actually defined!

  For $y_1(x)$,
  \begin{equation}
    b_n = \frac{a_0}{\prod_{k=1}^n(3k-1)(3k)}
  \end{equation}
  and so by the Ratio test
  \begin{align}
    R &= \frac{1}{\lim_{n\to\infty}\frac{|b_{n+1}|}{|b_n|}}\\
    &= \frac{1}{\lim_{n\to\infty}\frac{\prod_{k=1}^n(3k-1)(3k)}{\prod_{k=1}^{n+1}(3k-1)(3k)}}\\
    &= \frac{1}{\lim_{n\to\infty}\frac{1}{(3n-1)(3n)}}\\
    &= \infty.
  \end{align}
  Similarly, the radius of convergence of $y_2$ is also infinity. This means our power series converges everywhere and we have a solution on the whole real line.

  As a final remark, we can imagine expanding the solution around some other point $x_0 \neq 0$. If we do this, we get faster convergence around the new $x_0$, which is important for numerical applications. In addition, we can imagine situations in which $R < \infty$, in which case it would be necessary to find power series solutions around a collection of points; these solutions can in some cases be ``patched together'' to get a solution on a larger domain than just $[-R,R]$.

  \newpage

\item \underline{Theorem:} Consider the second order homogeneous ODE with variable coeffients
  \begin{equation}
    P(x)y'' + Q(x)y' + R(x)y = 0
  \end{equation}
  with standard form
  \begin{equation}
    y'' + p(x)y' + q(x)y = 0.
  \end{equation}
  We suppose $x_0$ is an ordinary point ($P(x) \neq 0$), and $p(x) = Q(x)/P(x)$, $q(x) = R(x)/P(x)$, and furthermore that $p$ and $q$ are analytic at $x_0$.

  Then
  \begin{enumerate}
  \item There exist power series solutions $y_1$ and $y_2$ such that $y = C_1y_1 + C_2y_2$ for all solutions $y$ defined at $x_0$;
  \item $\{y_1,y_2\}$ is a fundamental set;
  \item $R_j>0$, where $R_j$ is the radius of of convergence $y_j$ around $x_0$;
  \item $R_1,R_2 \geq \min\{|z-x_0|: z \in \CC, \,z\text{ are zeros of }P\}$.
  \end{enumerate}
  For example, in Airy's equation $P$ has no zeros. By part (d) of this theorem, that means $R_1,R_2 \geq \min \varnothing = \infty$, so $y_1$ and $y_2$ have infinite radius of convergence (as we found in the previous section).

  Note that to apply part (d), we have to search for zeros $z$ (where $P(z) = 0$) \emph{in the complex plane}.

  \underline{Example 1:} Consider the DE
  \begin{equation}
    (1-x)y'' - y' = 0,\quad\quad x_0 = 0
  \end{equation}
  Then $P(x) = 1-x$, $Q(x) = -1$, $R(x) = 0$. $P(x)$ has a single zero at $x = 1$, so the theorem says a fundamental set of power series solutions can be found that converge on the disk of radius 1 around 0. Let's verify this by solving the differential equation.
  \begin{align}
    v = y' &\implies (1-x)v' - v = 0\\
    &\implies \frac{dv}{v} = \frac{dx}{1-x}\\
    &\implies \ln|v| = -\ln|1-x| + C\\
    &\implies y' = v = \frac{C_1}{1-x},
  \end{align}
  so
  \begin{align}
    y(x) &= \int \frac{C_1}{1-x}dx\\
    &= -C_1\ln(1-x) + C_2\\
    &= C_1y_1(x) + C_2y_2(x)
  \end{align}
  where
  \begin{equation}
    \begin{cases}
      y_1 = -\ln(1-x) = \sum_{n=1}^\infty\frac{x^n}{n},\\
      y_2 = 1.
    \end{cases}
  \end{equation}
  By the ratio test, $R_1 = 1$ (the radius of convergence of $-\ln(1-x)$), and $R_2 = \infty$ because $y_2$ is constant. So both radii of convergence are $\geq 1$, as predicted.

  \underline{Example 2:} Consider
  \begin{equation}
    (1+3x^2)y'' - xy' + 4y = 0.
  \end{equation}
  $P(x) \neq 0$ when $x = 0$, which motivates taking $x_0 = 0$ again. We use the theorem to get a lower bound on the radius of convergence of power series solutions around $x_0$.

  First we find the zeros of $P(x) = 1 + 3x^2$. Setting $P(z) = 0$ implies $z^2 = -1/3$, so the zeros of $P$ are $z = \pm i/\sqrt{3}$. The theorem tells us that we can find power series solutions
  \begin{align}
    y(x) = \sum_{n=0}^\infty a_n(x-x_0)^n
  \end{align}
  with radius of convergence $R \geq \min|z-x_0| = \min\{|i/\sqrt{3} - 0|,|-i/\sqrt{3} - 0|\} = 1/\sqrt{3}$. Remember that this means the series converges for all $|x-x_0| < R$, i.e. when the \emph{strict} inequality holds. The radius of convergence says nothing about what happens when $|x-x_0| = R$ (whether the series converges absolutely, or whether it converges at all).

  \underline{Example 3:} We have a differential equation
  \begin{equation}
    y'' + xy' + 2y = 0
  \end{equation}
  with \emph{initial conditions}
  \begin{equation}
    y(0) = 1 \quad\quad y'(0) = -2.
  \end{equation}
  The theorem will give us a trick to determine some useful information about the solutions without finding them explicitly.

  Suppose $\phi(x)$ is a solution. Our task is to find $\phi''(0)$ and $\phi'''(0)$, because this will determine the second and third coefficients $a_2,a_3$ in the power series solution, and we already have the first two coefficients (by Taylor's formula (why?) the initial conditions give us $a_0 = 1$ and $a_1 = -2$).

  Let's see how this bootstrapping process to find the coefficients of $\phi$ works. First, we isolate the highest order term to get $\phi''(0)$ (which will give us $a_2$).
  \begin{align}
    y'' &= -xy' - 2y\\
    \implies \phi''(0) &= -0\phi'(0)- \phi(0) = -1.
  \end{align}
  With this information in hand, we can do the same for $\phi'''(0)$:
  \begin{align}
    y''' &= -y' - xy'' - 2y' = -xy'' - 3y'\\
    \implies \phi'''(0) &= -0\phi''(0) - 3\phi'(0) = 0 - (3)(-2) = 6.
  \end{align}
  In this special example, we did not need $\phi''(0)$ to determine $\phi'''(0)$, but you will notice that in general we may need the lower order derivatives to get the next one. This is why this process is sometimes known as bootstrapping.

\item Euler's equation (aka the Cauchy-Euler equation), is given for constants $a_0$, $a_1$, $a_2$ by
  \begin{align}
    a_2x^2\frac{d^2y}{dx^2} + a_1x^1\frac{dy}{dx} + a_0x^0y = 0,\quad(a_2 \neq 0)\quad\quad
    \implies P(x) = a_2x^2,\quad P(0) = 0.
  \end{align}
  This equation is an example of a general situation that we are going to study. This situation is when we have a second-order linear homogeneous ODE
  \begin{equation}
    P(x)y'' + Q(x)y' + R(x)y = 0
  \end{equation}
  where $x_0$ is a \textbf{singular point} (i.e. it is not an ordinary point, $P(x_0) = 0$), but it is a \textbf{regular} singular point.

  By dividing by $a_2x^2$, we can assume without loss of generality that we are dealing with the equation
  \begin{equation}
    \frac{d^2y}{dx^2} + \frac{a_1}{x}\frac{dy}{dx} + \frac{a_0}{x^2}y = 0
  \end{equation}
  in standard form (both equations have the same solutions).

\item Classification of singular points.

  \underline{Regular singular point}

  We say $x_0$ is a \textbf{regular singular point} of a differential equation in standard form ($y'' + p(x)y' + q(x)y = 0$) if both limits
  \begin{equation}
    \lim_{x\to x_0}(x-x_0)p(x), \quad \lim_{x\to x_0}(x-x_0)^2q(x)
  \end{equation}
  exist in $\RR$ (in other words, they are finite). This definition may remind you of the notion of a \emph{removable singular point} from complex function theory.

  For example, in Euler's equation we have $p(x) = a_1/x$ and $q(x) = a_0/x^2$, $x_0  = 0$. In this case $\lim_{x\to x_0}(x-x_0)p(x) = a_1$ and $\lim_{x\to x_0}(x-x_0)^2q(x) = a_0$.

  However, Euler's equation can be easily solved with methods from second year calculus. Our first real motivation for studying singular points will come from \textbf{Bessel's equation}. This is the equation
  \begin{equation}
    y'' + \frac{1}{x}y' + \frac{x^2-\nu^2}{x^2}y = 0
  \end{equation}
  where $\nu$ is a parameter. Bessel's equation also has a regular singular point at zero, and is much less tractable than Euler's equation with our current tools.

  \medskip

  \underline{Irregular singular point}

  If either of the limits above does not exist or blows up to $\infty$ or $-\infty$, then $x_0$ is called an \textbf{irregular singular point}.

  \medskip

  \underline{Example:} Solve the Euler equation
  \begin{equation}
    x^2\frac{d^2y}{dx^2} - 2x\frac{dy}{dx} - 4y = 0
  \end{equation}
  The idea behind Euler equations is the following observation. Notice that if we set $x = e^t$, then
  \begin{equation}
    x\frac{dy}{dx} = \frac{dx}{dt}\frac{dy}{dx} = \frac{dy}{dt}.
  \end{equation}
  This is a great simplification of the linear term in our differential equation! But what about the second order term? With a little more calculus, we find
  \begin{equation}
    x^2\frac{d^2y}{dx^2} = \frac{d^2y}{dt^2} - \frac{dy}{dt}
  \end{equation}
  This is fanastic, because it turns Euler's equation into a constant coefficient equation, which we know how to solve:
  \begin{equation}
    \left(\frac{d^2y}{dt^2} - \frac{dy}{dt}\right) - 2\frac{dy}{dt} - 4y = 0.
  \end{equation}
  This is the motivation for setting $x = e^t$.

  The rest of the solution is familiar; we find the characteristic polynomial $\tilde p(z) = z^2 - 3z - 4 = (z-4)(z+1)$, with roots $z = 4$ and $z = -1$. So we can write the general solution
  \begin{equation}
    y = C_1e^{4t}+C_2e^{-t}.
  \end{equation}
  But the original equation was not written in terms of $t$. Substituting back in $x = e^t$, we obtain the final solution in terms of $x$:
  \begin{equation}
    y(x) = C_1x^4+C_2x^{-1}.
  \end{equation}

\item More about singular points.

  We will do a few examples on how to find and classify singular points.

  \underline{Example:} Find the singular points of the following equation and classify them.
  \begin{equation}
    2x^2(1-x)y'' + 3xy' + (5-x)y = 0
  \end{equation}
  First we write down the coefficient polynomial $P(x) = 2x^2(1-x)$, and divide out to get
  \begin{equation}
    y'' + \frac{3x}{2x^2(1-x)}y' + \frac{5-x}{2x^2(1-x)}y = 0
  \end{equation}
  in standard form. The singular points are the zeros of $P$, which are $x_0 = 0$ and $x_0 = 1$.

  For $x_0 = 0$,
  \begin{align}
    \lim_{x\to x_0}(x-x_0)p(x) &= \lim_{x\to 0} x\frac{3x}{2x^2(1-x)} = \frac{3}{2} \in \RR\\
    \lim_{x\to x_0}(x-x_0)^2q(x) &= \lim_{x\to 0} x^2\frac{5-x}{2x^2(1-x)} = \frac{5}{2} \in \RR
  \end{align}
  both exist, so $x_0 = 0$ is a \underline{regular singular point.}

  For $x_0 = 1$,
  \begin{align}
    \lim_{x\to x_0}(x-x_0)p(x) &= \lim_{x\to 1} (x-1)\frac{3x}{2x^2(1-x)} = \lim_{x\to 1} \frac{-3}{2x} = -\frac{3}{2} \in \RR\\
    \lim_{x\to x_0}(x-x_0)^2q(x) &= \lim_{x\to 1} (x-1)^2\frac{5-x}{2x^2(1-x)} = \lim_{x\to 1} (1-x)\frac{5-x}{2x^2} = 0 \in \RR
  \end{align}
  so $x_0 = 1$ is also a regular singular point.

  \underline{Setting for the general case:}
  \begin{equation}\label{eq:general-second-order-homo}
    y'' + p(x)y' + q(x)y = 0, \quad\quad x\neq x_0,
  \end{equation}
  where $x_0$ is a regular singular point.

  We can do a clever manipulation (due to Frobenius) to make this equation look like Euler's equation. Multiplying by $x^2$, we get
  \begin{equation}
    x^2y'' + x[xp(x)]y' + [x^2q(x)]y = 0.
  \end{equation}
  We have suggestively factored the terms so that the coefficients of $y$ and $y'$ are exactly the functions we know the limit of as $x \to x_0$. (Look at the definition of a regular singular point). Since the limits exist, these coefficients
  \begin{equation}
    xp(x),\quad\quad\quad x^2q(x)
  \end{equation}
  are analytic functions, and they have power series
  \begin{align}
    xp(x) &= p_0 + p_1x + p_2x^2 + p_3x^3 + \dots\\
    x^2q(x) &= q_0 + q_1x + q_2x^2 + q_3x^3 + \dots
  \end{align}
  If we let $\hat p(x) = xp(x)$ and $\hat q(x) = x^2q(x)$, then our differential equation becomes
  \begin{align}
    x^2y'' + x\hat p(x)y' + \hat q(x)y = 0.
  \end{align}
  When $x$ is very close to $x_0$, $\hat p(x) \approx p_0$ and $\hat q(x) \approx q_0$, and we see that this equation is very nearly the same as the Euler equation
  \begin{align}
    x^2y'' + xp_0y' + q_0y = 0,
  \end{align}
  known as the \textbf{Euler counterpart} to the original equation.

  So what can we do with this? Since the differential equations are similar, we might imagine their solutions are also similar (and we already know the solutions to the Euler's equation). Assuming no repeated roots, the solutions to Euler's equation are just linear combinations of monomials. This line of thinking eventually leads to an ansatz for our original equation.

  \underline{Frobenius ansatz:} Solutions to (\ref{eq:general-second-order-homo}) (assuming no repeated roots in the associated characteristic polynomial) are of the form
  \begin{equation}
    y = x^r \quad\text{plus ``something''}
  \end{equation}

\end{enumerate}

\subsection{The method of Frobenius}

In this section we will make a synthesis of the theory of the previous section to present a general solution strategy for homogeneous ODE with polynomial coefficients.

\begin{enumerate}

\item On the Euler counterpart and the Frobenius ansatz.

  The Euler counterpart of (\ref{eq:general-second-order-homo}) is
  \begin{equation}
    x^2y'' + xp_0 y' + q_0y = 0,
  \end{equation}
  where $xp(x) = p_0 + p_1x + p_2x^2 + \dots$ and $x^2q(x) = q_0 + q_1x + q_2x^2 + \dots$

  Recall the general solution strategy for Euler's equation is making the substitution $x = e^t$, where one recovers the solutions $e^{rt} (= x^r)$, $r$ being a root of the characteristic polynomial of the constant coefficient equation in $t$ that Euler's equation reduces to. If characteristic polynomial has repeated roots, the second solution is given by $te^{rt} (= (\ln|x|)x^r)$. We will use this to formulate the Frobenius ansatz.

  \underline{Idea:} Let's just work with the $x^r$ solution (to the Euler counterpart) for now. The $(\ln|x|)x^r$ solution will come into play later.

  We come up with a candidate solution
  \begin{equation}\label{eq:frobenius-solution-formulation}
    y(x) = x^r\left(\sum_{n=0}^\infty a_nx^n\right)
    = \sum_{n=0}^\infty a_nx^{n+r},
  \end{equation}
  which takes the approximate solution $x^r$ and perturbes it by an analytic function that we solve for. Note that (\ref{eq:frobenius-solution-formulation}) is not in general a power series, because $r$ might not be an integer (or it might be negative)!

  \underline{Example:} Solve
  \begin{equation}\label{eq:example-frobenius-equation}
    2x^2y'' + 3xy' - (x+1)y = 0.
  \end{equation}
  In standard form, the equation is
  \begin{equation}
    y'' + \frac{3x}{2x^2}y' - \frac{x+1}{2x^2}y = 0.
  \end{equation}
  Is zero a regular point? Checking the two limits,
  \begin{align}
    \lim_{x\to0}xp(x) &= \lim_{x\to 0}x\frac{3x}{2x^2}\quad\text{ exists}\\
    \lim_{x\to0}x^2q(x) &= \lim_{x\to 0}x^2\frac{x+1}{2x^2}\quad\text{ exists}
  \end{align}
  so $x_0 = 0$ is a regular singular point. We need that it is regular for the Frobenius ansatz to apply.

  Our ansatz is
  \begin{equation}
    y(x) = \sum_{n=0}^\infty a_nx^{n+r}.
  \end{equation}
  We may assume $a_0 \neq 0$, for otherwise we could just shift $r$ by one (r and $a_k$ are undetermined coefficients), and get
  \begin{align}
    y(x) &= \sum_{n=1}^\infty a_nx^{n+r}\\
    &= \sum_{n=0}^\infty a_{n+1}x^{n+1+r}\\
    &= \sum_{n=0}^\infty b_nx^{n+1+r}.
  \end{align}
  You may then say ``well, why can't $b_0 = 0$?'' If $b_0 = 0$, then we repeat this process again and you might complain ``but now we could have $c_0 = 0$!''. But if we repeat this process for long enough, eventually one of the coefficients must be nonzero, otherwise they would all be zero and we would have the absurdity $y \equiv 0$!

  With that out of the way, we can assume $a_0 \neq 0$ and solve for all the coefficients. First of all, we have
  \begin{align}
    y(x) &= \sum_{n=0}^\infty a_nx^{r+n}\quad\quad(a_0 \neq 0)\\
    \implies y'(x) &= \sum_{n=0}^\infty a_n(r_n)x^{r+n-1}\\
    \implies y''(x) &= \sum_{n=0}^\infty a_n(r+n)(r+n-1)x^{r+n-2}
  \end{align}
  Applying these formulas to (\ref{eq:example-frobenius-equation}), we have
  \begin{align}
    0 &= 2x^2\sum_{n=0}^\infty a_n(r+n)(r+n-1)x^{r+n-2} + 3x\sum_{n=0}^\infty a_n(r+n)x^{r+n-1}
    - (x+1)\sum_{n=0}^\infty a_nx^{r+n}\\
    &= \sum_{n=0}^\infty a_n(r+n)(r+n-1)2x^{r+n} + \sum_{n=0}^\infty a_n(r+n)3x^{r+n}
    - \sum_{n=0}^\infty a_nx^{r+n+1} - \sum_{n=0}^\infty a_nx^{r+n+1}.
  \end{align}
  Rearranging and collecting terms (and using index shifting), we get
  \begin{align}
    0 &= \sum_{n=0}^\infty a_n\left(2(r+n)(r+n-1) + 3(r+n) - 1\right)x^{r+n}
    - \sum_{k=1}^\infty a_{k-1}x^{r+k}\\
    &= (2r(r-1) + 3r - 1)a_0x^r + \sum_{n=1}^\infty \left[a_n\left(2(r+n)(r+n-1) + 3(r+n) - 1\right) - a_{n-1}\right]x^{r+n}\\
    &= (2r(r-1) + 3r - 1)a_0x^r + \sum_{n=1}^\infty \left[\left(2(r+n)(r+n-1) + 3(r+n) - 1\right)a_n - a_{n-1}\right]x^{r+n}\\
    &= (2r^2 + r - 1)a_0x^r + \sum_{n=1}^\infty \left[\left(2(r+n)^2 + (r+n) - 1\right)a_n - a_{n-1}\right]x^{r+n}
  \end{align}
  where we have used $k = n+1$. This implies that all the coefficients of the $x^m$ are zero, which gives us the recurrence equation
  \begin{align}
    0 &= \left(2(r+n)^2 + (r+n) - 1\right)a_n - a_{n-1}\\
    \implies a_n &= \frac{a_{n-1}}{2(r+n)^2 + (r+n) - 1}
  \end{align}
  as well as the equation from the $x^r$ term:
  \begin{equation}\label{eq:r-equation-frob}
    2r^2 + r - 1 = 0
  \end{equation}
  since $a_0 \neq 0$. This is the famous \textbf{indicial equation} of Frobenius. Notice that $a_0$ is left as a free variable; it will become a coefficient in the general solution (remember we have not imposed any initial conditions thus far).

  \underline{Remark} The indicial equation (\ref{eq:r-equation-frob}) turns out to be precisely the \emph{characteristic equation for the Euler counterpart}. This fact is a general feature of this technique of Frobenius.

\item Summary on obtaining the indicial equation.

  Start with the differential equation
  \begin{equation}
    P(x)y'' + Q(x)y' + R(x)y = 0
  \end{equation}
  and formulate its Euler counterpart
  \begin{equation}
    x^2y'' + p_0xy' + q_0y = 0.
  \end{equation}
  Then apply the trial solution $y = x^r$ in the Euler counterpart, and you will get a polynomial equation in $r$. This is the indicial equation, and it corresponds to the characteristic equation of the Euler counterpart after substituting $x = e^t$ and rewriting it as a constant coefficient equation in terms of $t$.

  For instance, if the original equation is
  \begin{equation}
    2x^2y'' +3xy' - (1+x)y = 0
  \end{equation}
  as in our previous example, then the Euler counterpart is
  \begin{equation}
    2x^2y'' + 3xy' - y = 0
  \end{equation}
  (factors of $2$ don't matter).
  Trying $y(x) = x^r$ gives
  \begin{equation}
    (2r(r-1) + 3r - 1)x^r = 0
  \end{equation}
  and so the indicial equation is
  \begin{equation}
    2r^2 + r - 1 = (2r - 1)(r+1) = 0.
  \end{equation}

\item Using the indicial equation to find the general solution.

  Let's finish off our example. The roots of the indicial equation $r_1 = 1/2$, $r_2 = -1$ give us two possible solutions with the Frobenius ansatz, namely
  \begin{align}
    y_1(x) &= x^{1/2}\sum_{n=0}^\infty \alpha_nx^n\\
    y_2(x) &= x^{-1}\sum_{n=0}^\infty \beta_nx^n
  \end{align}
  Where $\alpha_n$ and $\beta_n$ are found by plugging into the original differential equation, but by assumption $\alpha_0 \neq 0$ and $\beta_0 \neq 0$. I claim $y_1$ and $y_2$ are linearly independent.

  To prove this, we need to show that the only solution to $c_1y_1(x) + c_2y_2(x) = 0$ that holds for all $x$ is $c_1 = c_2 = 0$. But
  \begin{align}
    c_1 y_1(x) + c_2y_2(x) &= 0\\
    \implies c_1x^{1/2}\sum_{n=0}^\infty\alpha_nx^n + c_2x^{-1}\sum_{n=0}^\infty\beta_nx^n &= 0\\
    \implies c_1x^{3/2}\sum_{n=0}^\infty\alpha_nx^n + c_2\sum_{n=0}^\infty\beta_nx^n &= 0.\label{eq:proving-linear-indep-in-frob}
  \end{align}
  Taking the limit as $x\to 0$ in the last equation, the only term that survives is $c_2\beta_0$. Since $\beta_0 \neq 0$, we must have $c_2 = 0$. But then plugging $c_2 = 0$ back into the equation, we have
  \begin{equation}\\
    c_1x^{3/2}\sum_{n=0}^\infty\alpha_nx^n = c_1(\alpha_0x^{3/2} + \alpha_1x^{5/2} + \dots) = 0
  \end{equation}
  As usual for power series, all of the terms of this series must be zero! (One way to think about it is that they are all linearly independent functions, so if a linear combination of them adds up to zero then all the coefficients must be zero.) Since $\alpha_0 \neq 0$, just looking at the first term of this series we can conclude that $c_1\alpha_0 = 0\implies c_1 = 0$ as well. We have been forced to conclude $c_1 = c_2 = 0$ as desired, so we are done. The two solutions we get from Frobenius are linearly independent.

  If you look at the argument we just made, you will see that in the general case, in place of (\ref{eq:proving-linear-indep-in-frob}) we will get
  \begin{align}
    c_1x^{r_1 - r_2}\sum_{n=0}^\infty\alpha_nx^n + c_2\sum_{n=0}^\infty\beta_nx^n = 0.
  \end{align}
  (by our choice of naming $r_1$ and $r_2$, we can assume without loss of generality that $r_1 - r_2 > 0$; the case $r_1 = r_2$, i.e. when the indicial equation has only one root, is another thing altogether that we will come back to in a future section). Again we send $x\to 0$ and obtain $c_2 = 0$ and consequently $c_1 = 0$. But here we have to be extremely careful about our assumptions. We assumed that the two Frobenius ansatz solutions $y_1$ and $y_2$ exist, \emph{with the first coefficient nonzero.} It may be that the recurrence relations one derives to determine the coefficients of $y_1$ and $y_2$ only have solutions when the $\alpha_0 = 0$ or $\beta_0 = 0$, violating our hypothesis.

  The moral of the story is that we have to settle for something weaker. The true result is that whenever $r_1 - r_2$ is \emph{not an integer}, then $y_1$ and $y_2$ as constructed from the Frobenius ansatz will be linearly independent (even if the differential equation forces $\alpha_0 = 0$ or $\beta_0 = 0$). We will revisit this result soon.

\item \underline{Summary of The General Result}

  We started with the problem of obtaining solutions to
  \begin{equation}
    P(x)y'' + Q(x)y' + R(x)y = 0
  \end{equation}
  by using power series around a \emph{regular singular point} $x_0 = 0$.

  First we converted the equation into standard form and multiplied by $x^2$:
  \begin{equation}
    x^2y'' + x(xp(x))y' + x^2q(x)y = 0
  \end{equation}
  where $p(x) = Q(x)/P(x)$ and $q(x) = R(x)/P(x)$.
  We expanded $xp(x)$ and $x^2q(x)$ as power series around $x_0$:
  \begin{align*}
    xp(x) &= p_0 + p_1x + p_2x^2 + \dots\\
    x^2q(x) &= q_0 + q_1x + q_2x^2 + \dots
  \end{align*}
  which gives us an Euler equation that is kind of a first approximation to the original equation, called the Euler counterpart:
  \begin{equation}
    x^2y'' + p_0xy' + q_0y = 0.
  \end{equation}

  Motivated by the form of solutions to the Euler counterpart, we introduced the Frobenius ansatz
  \begin{equation}
    y(x,r) = \sum_{n=0}^\infty a_n(r)x^{n+r},\quad\quad
    a_0(r) \neq 0
  \end{equation}
  (we can even assume $a_0(r) = 1$). Plugging the ansatz into the differential equation gives a consistency equation for $r$, the indicial equation
  \begin{equation}
    F(r) := r(r-1) + p_0 r + q_0 = 0
  \end{equation}
  ($F(r)$ is called the \textbf{indicial polynomial}). The roots of the indicial polynomial $r_1,r_2$ index two linearly independent solutions
  \begin{align}
    y_1(x) &= y(x,r_1) = \sum_{n=0}^\infty a_n(r_1)x^{n+r_1},\\
    y_2(x) &= y(x,r_2) = \sum_{n=0}^\infty a_n(r_2)x^{n+r_2}
  \end{align}
  \emph{provided $r_1 - r_2$ is not an integer}. $a_n(r_1)$ and $a_n(r_2)$ are just the undetermined coefficients we called $\alpha_n$ and $\beta_n$ earlier.

  This is what we had. And it works when $F(r)$ has two distinct roots. But what if it only has one root? This is a problem we will extend our method to solve in the future. Before we do that, we will elaborate on the other special case where the above fails; when $r_1 - r_2$ is an integer.

  \medskip

\item Suppose $r_1-r_2$ is an integer, say $r_1 - r_2 = -N \in \ZZ$. (The negative sign is for convenience later.)

  We show that $y_1$ and $y_2$ are not in general linearly independent. We have that $c_1y_1 + c_2y_2 = 0$ takes the following form:
  \begin{equation}
    c_1\sum_{n=0}^\infty \alpha_n x^{r+n} + c_2 \sum_{n=N}^\infty\beta_n'x^{r+n} = 0
  \end{equation}
  where we have set $r = r_1$, and $\beta_n' = \beta_{n - N}$. Assume $N \geq 0$. We can rewrite the above as
  \begin{equation}\label{eq:failure-to-be-lin-indep-frob}
    c_1\sum_{n=0}^{N-1}\alpha_nx^{r+n} + \sum_{n=N}^\infty (c_1\alpha_n + c_2\beta_n')x^{r+n} = 0
  \end{equation}
  Now it is clear that as long as $\alpha_0 = \alpha_1 = \dots = \alpha_{N-1} = 0$, it can be the case that $c_1$ and $c_2$ are both nonzero and the $\alpha_n$ and $\beta_n'$ terms simply cancel out. Since we cannot conclude $c_1 = c_2 = 0$, we cannot conclude that $y_1$ and $y_2$ are linearly independent (and in general they need not be).

  This failure to give linearly independent solutions does have a resolution, but it requires more advanced theory that we will not get into at the present moment. Remember, though, that at least $y_1$ and $y_2$ are linearly independent whenever $r_1 - r_2$ is not an integer. Hopefully you can see that in this case, the two sums cannot be ``overlapped'' as in (\ref{eq:failure-to-be-lin-indep-frob}). No power of $x$ has coefficients in both $y_1$ and $y_2$, so the coefficients can never cancel each other out as we just imagined, and the solutions are therefore always linearly independent for $r_1 - r_2 \notin \ZZ$.

\end{enumerate}

\section{Qualitative Methods}

\subsection{Some vocabulary}

We are going to start our study of qualitative methods by considering the idea of critical points. We have a system of ODE
\begin{equation}\label{eq:simple-constant-linear-homogeneous}
  \dot x = Ax\quad\quad
  A \in M_{2\times2}(\RR),\quad
  x = x(t)
\end{equation}
Consider all the times $t^*$ such that the velocity at $t^*$ is zero: $\dot x(t^*) = 0$. A \textbf{critical point} is the value $x^* = x(t^*)$ of the position at any of the ``critical'' times $t^*$. An \textbf{equilibrium solution} is a solution of the form $x(t) \equiv x^*$ for some critical point $x^*$. Clearly all the constant solutions are of this form, because constancy means $\dot x(t) = 0$.

\underline{Example:} The system of ODE
\begin{equation}
  \begin{cases}
    \dot x_1 = -x_1\\
    \dot x_2 = -2x_2
  \end{cases}
\end{equation}
has the critical point $x_1 = x_2 = 0$, so $x(t) = [0, 0]^T$ is an equilibrium solution.

\underline{Observation:} $A 0 = 0$ means $0$ is always a critical point to (\ref{eq:simple-constant-linear-homogeneous}).

In our study we will have to consider various cases depending on the eigenvalues of $A$. The possible situations are
\begin{enumerate}
\item Real, distinct eigenvalues of the same sign
\item Real, distinct eigenvalues of opposite sign
\item Equal (only one eigenvalue)
\item Complex $\alpha \pm i\beta$ ($\alpha \neq 0$, $\beta > 0$)
\item Complex $\pm i\beta$ ($\beta > 0$)
\end{enumerate}

\noindent
\underline{Case 1 (same sign):} We can write $A = P\Lambda P^{-1}$, where $p = [\xi^1, \xi^2]$ and $\Lambda = \text{diag}(\lambda_1, \lambda_2)$. The general solution to (\ref{eq:simple-constant-linear-homogeneous}) is
\begin{equation}\label{eq:general-solution-nodal-source}
  x(t) = c_1e^{\lambda_1t}\xi^1 + c_2e^{\lambda_2t}\xi^2.
\end{equation}

Drawing the direction field of (\ref{eq:simple-constant-linear-homogeneous}) shows that if $\lambda_1\neq0, \,\lambda_2\neq0$ then the \emph{only} equilibrium point is the origin. (Why?) In particular, if the eigenvalues are both positive, all trajectories flow away from the origin. In this case we say that the origin is a \textbf{nodal source}.

If both eigenvalues are negative we still have the general solution (\ref{eq:general-solution-nodal-source}), but now all solution curves are attracted to the origin. The origin is then called a \textbf{nodal sink}.

\noindent
\underline{Case 2 (opposite sign):} Suppose without loss of generality $\lambda_1 < 0 < \lambda_2$. Then the general solution is still
\begin{equation}\label{eq:general-solution-opposite-sign}
  x(t) = c_1e^{\lambda_1t}\xi^1 + c_2e^{\lambda_2t}\xi^2,
\end{equation}
but the behaviour is a bit different. Drawing the direction field shows that along the line $s\xi^1,\,s\in\RR$ the solution curves are attracted to the origin, but along the line spanned by $\xi^2$ the solution curves diverge. By drawing some more general solution curves, one finds that \emph{any} curve that starts off of the $s\xi^1,\,s\in\RR$ line eventually diverges to infinity, while the curves that start on that line follow it to the origin. This is unstable behaviour, and the origin is called a \textbf{saddle point}.

\noindent
\underline{Case 3 (equal eigenvalues):} Recall from linear algebra that if two eigenvectors $\xi^1$ and $\xi^2$ correspond to different eigenvalues, they are guaranteed to be linearly independent. Here the situation is a little more subtle. Let's suppose first that $\xi^1$ and $\xi^2$ are linearly independent eigenvectors corresponding to the one unique eigenvalue $\lambda$. The general solution is
\begin{align}
  x(t) &= c_1e^{\lambda t}\xi^1 + c_2e^{\lambda t}\xi^2\\
  &= e^{\lambda t}(c_1 \xi^1 + c_2\xi^2)
  &= e^{\lambda t}x(0).
\end{align}
Notice that \emph{all} starting points $x(0)$ are eigenvalues of $A$. If we draw the direction field, we see that the solution curves are all straight lines through the origin. We call the origin a \textbf{star node}.

If there do \emph{not} exist two linearly independent eigenvectors $\xi^1,\,\xi^2$, then we just have one eigenvalue $\xi$. When we solved this problem back in \S\ref{sec:first-order}, we found the solution
\begin{equation}
  x(t) = c_1e^{\lambda t}\xi + c_2e^{\lambda t}(\xi t + \eta)
\end{equation}
where $\eta$ was called the \emph{generalised eigenvector}, and satisfies
\begin{equation}
  \xi = (A - \lambda I)\eta
\end{equation}
The solutions are no longer just exponentials, but involve a linear polynomial times an exponential. Since $\eta \neq \xi$, the solution curves are

(1) straight lines if and only if $c_2 = 0$;

(2) $x(t) \sim c_2 te^{\lambda t} \xi$ as $t\to\pm\infty$ whenever $c_2 \neq 0$ (the linear part of $t\xi + \eta$ dominates)

In this case the origin is called a \textbf{degenerate node} or \textbf{improper node}. If $\lambda < 0$, it is a \textbf{degenerate sink}; if $\lambda > 0$ it is a \textbf{degenerate source}.

\bigskip

\noindent
\underline{Case 4 ($\lambda = \alpha \pm i\beta$, $\alpha \neq 0$, $\beta > 0$):} The eigen\emph{vectors} look like $\xi = p + iq$, where $p$ and $q$ are real vectors. For a given eigenvalue $\lambda$, the other eigenvalue is $\bar\lambda$. Similarly, taking the complex conjugate of $A\xi = \lambda \xi$ shows that the other eigenvector is $\bar\xi$ (remember that $A$ is real). Then the complex general solution to (\ref{eq:simple-constant-linear-homogeneous}) is
\begin{align}
  x(t) &= c_1e^{\lambda t}\xi + c_2e^{\bar\lambda t}\bar\xi\\
  &= c_1e^{\alpha t + i\beta t}(p + iq) + c_2e^{\alpha t - i\beta t}(p - iq)
\end{align}
Since we know a real trajectory will remain real under (\ref{eq:simple-constant-linear-homogeneous}), taking real and imaginary parts give the real general solutions
\begin{align}
  x(t) &= (c_1 + c_2)e^{\alpha t}(p\cos(\beta t) - q\sin(\beta t))
  + (c_1 - c_2)e^{\alpha t}(p\sin(\beta t) + q\cos(\beta t))\\
  &=: c_1'e^{\alpha t}(p\cos(\beta t) - q\sin(\beta t))
  + c_2'e^{\alpha t}(p\sin(\beta t) + q\cos(\beta t))
\end{align}
(remember $p$ and $q$ are vectors). Going back to the equations
\begin{align}\label{second-order-expanded-system}
  \dot x_1 &= a_{11}x_1 + a_{22}x_2\\
  \dot x_2 &= a_{21}x_1 + a_{12}x_2
\end{align}
(just the expanded form of $\dot x_1 = Ax$), it is useful for the qualitative analysis to use polar coordinates
\begin{equation}
  \begin{cases}
    x_1 = r\cos\theta\\
    x_2 = r\sin\theta
  \end{cases}
\end{equation}
where $r = r(t)$ and $\theta = \theta(t)$ are functions of $t$. Plugging these equations into (\ref{second-order-expanded-system}) gives an equation for $r$.

\underline{Equation for $r$}
\begin{equation}
  r^2 = x_1^2 + x_2^2
\end{equation}
\begin{equation}
  2r\frac{dr}{dt} = 2x_1\frac{dx_1}{dt} + 2x_2\frac{dx_2}{dt}
\end{equation}
which implies
\begin{align}
  \frac{dr}{dt} &= \frac{x_1}{r}\frac{dx_1}{dt} + \frac{x_2}{r}\frac{dx_2}{dt}\\
  &= \frac{x_1}{r}(a_{11}x_1 + a_{12}x_2) + \frac{x_2}{r}(a_{21}x_1 + a_{22}x_2)\\\label{eq:polar-expansion-for-r}
  &= r\cos\theta(a_{11}\cos\theta + a_{12}\sin\theta) + r\sin\theta(a_{21}\cos\theta + a_{22}\sin\theta).
\end{align}
To simplify this complicated expression, we can now we use the change of coordinates given by
\begin{equation}
  P^{-1}AP =
  \begin{pmatrix}
    \alpha & -\beta\\
    \beta & \alpha
  \end{pmatrix}
\end{equation}
for some nonsingular matrix $P$. Specifically, our original system $\dot x = Ax$ becomes
\begin{equation}
  P^{-1}\dot x = (P^{-1}AP)P^{-1}x
\end{equation}
or
\begin{equation}
  \dot y = (P^{-1}AP)y,
\end{equation}
where $y = P^{-1}x$.
This means we can assume without loss of generality $A$ is of the form
\begin{equation}
  A =
  \begin{pmatrix}
    \alpha & -\beta\\
    \beta & \alpha
  \end{pmatrix},
\end{equation}
and then multiply on the left by $P$ when we are done. By the symmetry of this \textbf{canonical form} of $A$, (\ref{eq:polar-expansion-for-r}) becomes
\begin{equation}
  \frac{dr}{dt} = \alpha r\cos^2\theta + \alpha r\sin^2\theta
\end{equation}
gives the final equation for $r$:
\begin{equation}
  \frac{dr}{dt} = \alpha r.
\end{equation}

\underline{Equation for $\theta$}
If
\begin{equation}
  A =
  \begin{pmatrix}
    \alpha & -\beta\\
    \beta & \alpha
  \end{pmatrix},
\end{equation}
then
\begin{equation}
  \frac{d\theta}{dt} = \beta
\end{equation}

\underline{Summary} (Under the canonical form):
\begin{equation}
  r(t) = r(0)e^{\alpha t},\quad\quad\theta(t)=\theta(0) + \beta t
\end{equation}
\begin{equation}
  x_1(t) = r(0)e^{\alpha t}\cos(\theta(0) + \beta t),\quad\quad
  x_2(t) = r(0)e^{\alpha t}\sin(\theta(0) + \beta t),\quad\quad
\end{equation}

The origin is an \textbf{asymptotically stable spiral point} if $\alpha < 0$ and an \textbf{unstable spiral source} if $\alpha > 0$.

\bigskip

\noindent
\underline{Case 5 ($\lambda = \pm i\beta$, $\beta> 0$):}
In this case the matrix $A$ has a canonical form
\begin{equation}
  A =
  \begin{pmatrix}
    0 & -\beta\\
    \beta & 0
  \end{pmatrix}
\end{equation}
The solutions are the same as for Case 4, except there is no divergence from the origin (the origin is no longer a source or a sink) and the trajectories are all circles around origin. In particular, no solution will converge to a point or diverge to infinity. In this case, the origin is called a \textbf{center}.

\underline{If $A$ is not in canonical form already...} Recall that we set $y = P^{-1}x$, so $x = Py$. Under this transformation back to the original coordinates, the solution curves may not exactly be circles, but the overall behaviour is the same. To see this, look at the radius $R^2 = x^T x = y^T P^TP y$. Expanding out into coordinates shows that this is in general the equation for an \emph{ellipse.}

\bigskip

\subsection{General setup}\label{sec:critical-points-general-setup}

In this section we will consider a general first order autonomous system
\begin{equation}\label{eq:first-order-autonomous}
  \dot x = f(x).
\end{equation}
In general, $f$ may be nonlinear.

We have already introduced the notions of \emph{critical points} $x_0$ where $f(x_0) = 0$, and the corresponding \emph{equilibrium solutions} $x(t) \equiv x_0$.

\begin{enumerate}
\item Example (oscillating pendulum):

  From physics, the equation of motion for a simple pendulum with damping is
  \begin{equation}\label{eq:oscillating-pendulum-eq-of-motion}
    mL^2\frac{d^2\theta}{dt^2} = -cL\frac{d\theta}{dt} - mgL\sin\theta
  \end{equation}
  Rewriting the equation in standard form, and replacing some of the constants with $\gamma$ and $\omega$, we obtain the equation
  \begin{equation}
    \frac{d^2\theta}{dt^2} = -\gamma\frac{d\theta}{dt} - \omega^2\sin\theta.
  \end{equation}
  To turn this into a first order system, we make the definitions $x = 0$, $\ds y = \frac{d\theta}{dt}$, which allows us to write down
  \begin{equation}\label{eq:oscillating-pendulum-first-order-system}
    \begin{cases}
      \frac{dx}{dt} = y\\
      \frac{dy}{dt} = -\gamma y - \omega^2\sin x
    \end{cases}
  \end{equation}
  Notice that this system is nonlinear, because of the $-\omega^2\sin x$ term.

  \underline{Find critical points.} Setting the right hand side of (\ref{eq:oscillating-pendulum-first-order-system}) to zero gives us the equations $y = 0$ and $\sin x = 0$. So the critical points are
  \begin{equation}\label{eq:oscillating-pendulum-crit-points}
    (x,y) = (n\pi,0)\quad n \in \ZZ
  \end{equation}

  Now we think about stability. Consider the $(x,y) = (0,0)$ critical point / equilibrium solution. Thinking about the physics of the pendulum (this is the stationary solution where the pendulum is hanging straight down), it is easy to see that this is a stable critical point. An even stronger property holds, which we will define in a minute; $(0,0)$ is going to be an \textbf{asymptotically stable critical point}.

  Similarly, thinking about $x = \pi$ (the pendulum is upside down), we have that $(x,y)=(\pi,0)$ is an unstable critical point. (And notice that all critical points are one of these two cases, since $\theta$ is $2\pi$ periodic.)

  \medskip

\item Stable and asymptotically stable critical points.

  A \textbf{stable critical point} is a point $x_0$ where if we perturb the initial condition $x(0) \approx x_0$, the solution $x(t)$ remains close to the equilibrium solution $x(t) \approx x_0$ for all $t$.

  The other case is the \textbf{asymptotically stable critical point}. $x_0$ is said to by asymptotically stable if the solutions $x(t)$ just considered do not merely remain close to $x_0$ but actually converge:
  \begin{equation}
    x(0) \approx x_0 \implies \lim_{t\to\infty}x(t) = x_0
  \end{equation}

  Asymptotically stable implies stable, but the converse is not true in general.

  \medskip

\item Basins of attraction.

  We are looking at the collection of all initial points for which the solution curves converge to a given asymptotically stable point. Formally, if $x_0$ is an asymptotically stable point, then its \textbf{basin of attraction} is the set
  \begin{equation}\label{eq:basin-of-attraction}
    \left\{ x(0) \in \RR^2 \big| \lim_{t\to\infty}x(t) = x_0\right\}.
  \end{equation}
  For example, if the eigenvalues of $A$ are real and negative, then $(0,0)$ is a nodal sink of $\dot x = Ax$, and has a basin of attraction $\RR^2$. More generally, if $\Re(\lambda) < 0$ for each eigenvalues $\lambda$, then the origin is a spiral sink and again $\text{Basin} = \RR^2$.

  \underline{Example (nonlinear example)} Consider the system
  \begin{equation}
    \begin{cases}
      \dot x = -x\\
      \dot y = y - y^3
    \end{cases},
    \quad\quad\quad (x,y) \in \RR^2.
  \end{equation}
  We want to find the critical points and the associated basins of attraction. To obtain the critical points, we set the right hand side to zero:
  \begin{equation}
    \begin{cases}
      -x = 0\\
      y - y^3 = 0
    \end{cases}
    \implies E_1 = (0,0),\quad E_2 = (0,1),\quad E_3 = (0, -1)
  \end{equation}
  ($E_1,E_2,E_3$ are the critical points).

  To find the basins of attraction, we need to analyse the origin equations. We have
  \begin{equation}
    \dot x = -x \implies x(t) = x(0)e^{-t}
  \end{equation}
  and the equation $\dot y = y(1-y^2)$. Looking at a (one dimensional) phase diagram for $\dot y$, we have that $y = 0$ is an unstable source, and $y = \pm 1$ are asymptotically stable sinks. Since $x(t) \to 0$ as $t \to \infty$, we have that $E_1$ is not stable (so it does not have a basin) but $E_2$ and $E_3$ are asymptotically stable.

  For $E_2 = (0,1)$, any initial $x$ will give a solution converging to $x = 0$. But for $y$, the phase diagram in $y$-space shows that for convergence to $y = 1$, we must have $y(0) > 0$ (otherwise $y(t)$ will either sit at $y = 0$ or converge to $y = -1$). So its basin is $\{ (x,y) \in \RR^2 | y > 0\}$.

  For $E_3 = (0,-1)$, we must have $y(0) < 0$ for convergence, and so its basin is $\{(x,y)\in \RR^2 | y < 0\}$.

  Notice that there is a line $\{ (x,0) | x \in \RR\} \subset \RR^2$ that belongs to neither basin of attraction, but separates the two basins. This line is called the \textbf{separatrix}.

  \medskip

  \underline{Example} (Two linearly independent vectors but only one eigenvalue.)

  Consider
  \begin{equation}
    \dot x = P
    \begin{pmatrix}
      \lambda & 0\\
      0 & \lambda
    \end{pmatrix}
    P^{-1} x.
  \end{equation}
  In order to work with the diagonal system, we should change coordinates $y = P^{-1}x$. Now we can just work with
  \begin{equation}
    \dot y =
    \begin{pmatrix}
      \lambda & 0\\
      0 & \lambda
    \end{pmatrix}
    y.
  \end{equation}
  This decouples the system, so that it suffices to consider (using $y = [x_1,x_2]^T$)
  \begin{equation}
    \begin{cases}
      \frac{dx_1}{dt} = \lambda x_1\\
      \frac{dx_1}{dt} = \lambda x_1
    \end{cases}
    \implies\quad
    \frac{dx_2}{dx_1} = \frac{dx_2/dt}{dx_1/dt}
    = \frac{\lambda x_2}{\lambda x_1}
    = \frac{x_2}{x_1}.
  \end{equation}

  More generally,
  \begin{equation}
    \begin{cases}
      \frac{dx_1}{dt} = F(x_1,x_2)\\
      \frac{dx_1}{dt} = G(x_1,x_2)
    \end{cases}
    \implies\quad
    \frac{dx_2}{dx_1} = \frac{dx_2/dt}{dx_1/dt}
    = \frac{G(x_1,x_2)}{F(x_1,x_2)}.
  \end{equation}
  We have come up with a new differential equation independent of time parameter (so we can forget about it). Generally, we can solve this equation using methods from basic ODE theory (e.g. it's often separable, like in the special case above). We get solutions of the form $H(x_1,x_2) = c$. For example, in the special case $F/G = x_2 / (-x_1)$, we have $-x_2dx_2 = x_1dx_1$, and integrating both sides we get $x_1^2 + x_2^2 = c$. (The solutions are circles.) On the other hand, the matrix that generates this system is
  \begin{equation}
    A =
    \begin{pmatrix}
      0 & -1\\
      1 & 0
    \end{pmatrix},
  \end{equation}
  and we saw the circular solution curves for this system before.

  \medskip

\item General setting.

  We can write a generic first order system of two dimensions as
  \begin{equation}\label{eq:general-first-order-system}
    \begin{cases}
      \frac{dx}{dt} = F(x,y)\\
      \frac{dy}{dt} = G(x,y)
    \end{cases}
  \end{equation}
  where $F$ and $G$ are arbitrary functions of $x$ and $y$. In order to apply our framework for the analysis of critical points, we will make two basic assumptions.

  \underline{Assumption 1:} $(x_0,y_0)$ is a critical point of (\ref{eq:general-first-order-system})

  \underline{Assumption 2:} $F$ and $G$ are continuously differentiable, or \emph{of class} $C^1$, in a neighbourhood of $(x_0,y_0)$.

  The purpose of Assumption 2 is to allow us to use Taylor's approximation theorem (or in this case, just the definition of the derivative of a multivariate function) to write
  \begin{align}
    F(x,y) &= F(x_0,y_0) + F_x(x_0,y_0)(x-x_0) + F_y(x_0,y_0)(y-y_0) + \eta_1(x-x_0,y-y_0)\\
    G(x,y) &= G(x_0,y_0) + G_x(x_0,y_0)(x-x_0) + G_y(x_0,y_0)(y-y_0) + \eta_2(x-x_0,y-y_0)
  \end{align}
  where subscripts denote partial derivatives, and the error term $\eta = (\eta_1,\eta_2)$ satisfies the bound
  \begin{equation}
    \lim_{(x,y)\to(x_0,y_0)}\frac{|\eta(x-x_0,y-y_0)|}{|(x,y)-(x_0,y_0)|} = 0.
  \end{equation}
  These expressions in turn motivate the introduction of some new variables $u = x-x_0$, $v = y-y_0$, and $w = (u,v)$.

  We can then rewrite the system (\ref{eq:general-first-order-system}) in the following form:
  \begin{equation}
    \begin{pmatrix}
      \frac{dx}{dt} \\ \frac{dy}{dt}
    \end{pmatrix}
    =
    \begin{pmatrix}
      \frac{du}{dt} \\ \frac{dv}{dt}
    \end{pmatrix}
    =
    \begin{pmatrix}
      F_x & F_y\\
      G_x & G_y
    \end{pmatrix}
    \begin{pmatrix}
      u \\ v
    \end{pmatrix}
    +
    \begin{pmatrix}
      \eta_1(w) \\ \eta_2(w)
    \end{pmatrix}
  \end{equation}
  with $\eta = (\eta_1,\eta_2)$ satisfying
  \begin{equation}
    \lim_{w\to 0}\frac{|\eta(w)|}{|w|} = 0
  \end{equation}
  In this form we can recognise the matrix as simply the Jacobian $J(x,y)$ of the coordinate transformation $(x,y)\mapsto (F,G)$.

\medskip

\item Locally linear systems.

  We say that we have a \textbf{locally linear system} if the Jacobian is invertible ($\det J(x_0,y_0) \neq 0$). The \underline{key point} of this definition is that it ensures $(x_0,y_0)$ is an isolated critical point. To see where this can fail, consider the following example.

  \underline{Example (Bad case):}
  \begin{equation}
    \begin{cases}
      x' = x^2\\
      y' = 0
    \end{cases}
  \end{equation}
  Setting the right hand side to zero, we find that the set of critical points is just the $y$-axis $\{(0,y)|y \in \RR\}$. When we compute the Jacobian:
  \begin{equation}
    J(x,y) =
    \begin{pmatrix}
      2x & 0\\
      0 & 0
    \end{pmatrix}
  \end{equation}
  we see that $J(0,0) = 0$ and so the system is not locally linear around $(x_0,y_0) = (0,0)$. In this case, the failure of the invertibility of the Jacobian comes from the fact that the origin is not an isolated critical point.

  \medskip
  The reason we want isolated singularities is so that we can use the following.\\
  \underline{Fact:} Stability results from the corresponding linear system
  \begin{equation}
    \begin{pmatrix}
      \dot u \\ \dot v
    \end{pmatrix}
    =
    J(x_0,y_0)
    \begin{pmatrix}
      u \\ v
    \end{pmatrix}
  \end{equation}
  carry over to the locally linear system
  \begin{equation}
    \begin{pmatrix}
      \dot x \\ \dot v
    \end{pmatrix}
    =
    \begin{pmatrix}
      F(x,y) \\ G(x,y)
    \end{pmatrix}
  \end{equation}
  except for the case where the eigenvalues are purely imaginary. To see why the purely imaginary case is special, let's look at some special cases.

  \underline{Example ($\lambda = \pm i\beta$)}\\
  1.
  \begin{equation}
    \begin{cases}
      x' = y\\
      y' = -x
    \end{cases}
    \implies\quad
    A =
    \begin{pmatrix}
      0 & 1\\
      -1 & 0
    \end{pmatrix}
  \end{equation}
  The characteristic polynomial of $A$ is $\lambda^2 + 1$, so its eigenvalues are $\lambda = \pm i$. As we have seen before, the origin is a stable but not asymptotically stable critical point (it is a centre).

  2.
  \begin{equation}
    \begin{cases}
      x' = y - x(x^2 + y^2)\\
      y' = -x - y(x^2 + y^2)
    \end{cases}
  \end{equation}
  The origin is still a critical point, and the Jacobian is
  \begin{equation}
    J(x,y) =
    \begin{pmatrix}
      -3x^2 - y^2 & 1 - 2xy\\
      -1-2xy & -x^2-3y^2
    \end{pmatrix}.
  \end{equation}
  At the origin, the Jacobian is precisely $A$: $J(0,0)  = A$. So the \emph{linearised} system is the one presented just above.

  However, the behaviour of (2) is not fully captured by the linearised system. To see this, we move to polar coordinates. Differentiating $r^2 = x^2 + y^2$ with respect to $t$, we get
  \begin{equation}
    \begin{split}
      2r\frac{dr}{dt} &= 2x\frac{dx}{dt} + 2y\frac{dy}{dt}\\
      &= 2x(y - x(x^2+y^2)) + 2y(-x-y(x^2+y^2))\\
      &= -2(x^2+y^2)^2\\
      &= -2r^4.
    \end{split}
  \end{equation}
  This means $\ds\frac{dr}{dt} = -r^3 < 0$ for all $r$, so every solution curve converges to the origin $r=0$! Unlike in its linearised version, the origin is an asymptotically stable point of (2).

  Now let's consider the very closely related\\
  3.
  \begin{equation*}
    \begin{cases}
      x' = y + x(x^2+y^2)\\
      y' = -x + y(x^2+y^2)
    \end{cases}.
  \end{equation*}
  Again it is easy to see that the corresponding linear system is given by (1) above (all we did was flip the sign of the nonlinear terms). However, this system displays even more disparate behaviour. By the same analysis as above, we find that $\ds\frac{dr}{dt} = r^3 > 0$ for all $r$, and hence the nonzero solution curves all diverge to infinity. In this case, the origin as an unstable source.

  The message to take away from this example is while the behaviour of linearised systems generally carries over to that of the nonlinear system, it tells one absolutely nothing if the critical point is a centre of the linear system (the coefficient matrix has purely imaginary eigenvalues).

  \medskip

\item Example: analysis of critical points using $J(x,y)$.

  Consider the system
  \begin{equation}\label{eq:crit-pt-jacobian-system}
    \begin{cases}
      \frac{dx}{dt} = y\\
      \frac{dy}{dt} = -x+x^3
    \end{cases}.
  \end{equation}

  \underline{Step 1:} Find critical points.

  Setting the right hand side to zero, we have that all critical points satisfy $y = 0$ and hence $x(1-x^2) = 0$. This gives us the three critical points
  \begin{equation}
    P_1 = (0,0), \quad\quad
    P_2 = (0,1), \quad\quad
    P_3 = (0,-1).
  \end{equation}

  \underline{Step 2:} Calculate $J(x,y)$.

  Differentiating the right hand side of (\ref{eq:crit-pt-jacobian-system}) with respect to the various variables, we get
  \begin{equation}
    J(x,y) =
    \begin{pmatrix}
      0 & 1\\
      -1+3x^2 & 0
    \end{pmatrix}.
  \end{equation}

  \underline{Step 3:} Evaluate $J$ at the various critical points and analyse the behaviour of the corresponding linear systems.

  For the first critical point, we have
  \begin{equation}
    J(P_1) =
    \begin{pmatrix}
      0 & 1\\
      -1 & 0
    \end{pmatrix}.
  \end{equation}
  We have seen already that the eigenvalues of $J(P_1)$ are purely imaginary, so we sadly cannot conclude anything about the behaviour of the nonlinear system around $P_1$.

  For the second critical point, we have
  \begin{equation}
    J(P_2) =
    \begin{pmatrix}
      0 & 1\\
      2 & 0
    \end{pmatrix}.
  \end{equation}
  The characteristic polynomial of $J(P_2)$ is $\lambda^2 - 2$, and so the matrix has eigenvalues $\lambda = \pm\sqrt{2}$. By our theory of the classification of critical points of linear systems, we conclude that the linearised system has $P_2$ as an unstable saddle point. Since the general system is locally linear ($J$ is invertible at $P_2$), and the eigenvalues are not purely imaginary, $P_2$ is an unstable saddle point for the original system as well.

  Finally, the third critical point has $J(P_3) = J(P_2)$, so $P_3$ is also an unstable saddle point.

  \bigskip

\end{enumerate}

\subsection{Lyapunov method}

The Lyapunov method is a way of classifying critical points with the use of a distinguished real-valued function of $x$ and $y$, known as the Lyapunov function. The inspiration for the Lyapunov function comes from certain properties of the total energy function in classical mechanics systems (like the mass-spring system) that make it useful for the classification of critical points. In general, a Lyapunov function will be chosen specifically for the system at hand, and there is no general formula to produce a suitable function.

\begin{enumerate}

\item \underline{Example:} Consider the mass-spring system
  \begin{equation}
    m\ddot  + c\dot x + kx = 0
  \end{equation}
  \begin{equation}
    \implies \begin{cases}
      \dot x = y\\
      \dot y = - \frac{k}{m}x - \frac{c}{m}y
    \end{cases},
  \end{equation}
  where $y = \dot x$.
  The total mechanical energy is given by
  \begin{align}
    V(x,y) &= KE + PE\\
    &= \frac{1}{2}m\dot x^2 + \int_0^x(-F_{\text{spring}}(x_1))dx_1\\
    &= \frac{1}{2}my^2 + \int_0^xkx_1dx_1\\
    &= \frac{m}{2}y^2 + \frac{k}{2}x^2.
  \end{align}
  The function $V(x,y)$, in dynamical systems, is called a \textbf{Lyapunov function}. Notice that $V$ is nonnegative everywhere, and is strictly positive away from the origin:
  \begin{equation}
    V(0,0) = 0,\quad\quad\quad
    V(x,y) > 0 \quad\forall (x,y) \neq (0,0).
  \end{equation}
  Remembering that $y = \dot x$, we see that $(0,0)$ is an equilibrium point; the mass does not move after being placed at $x = 0$.

\item More generally, the Lyapunov function gives us what is known as a zeroth order test for critical points of general systems.

  Assume
  \begin{equation}
    \begin{cases}
      \dot x = F(x,y)\\
      \dot y = G(x,y)
    \end{cases},
    \quad\quad F(0,0) = G(0,0) = 0.
  \end{equation}
  Then $(0,0)$ is a critical point. We also assume that there exists a (differentiable) function $V(x,y)$ such that $V(0,0) = 0$.

  There are four possible conditions on the function $V$:
  \begin{enumerate}
  \item $V$ is said to be \textbf{positive definite} if $$V(x,y) > 0\text{ for all }(x,y) \neq (0,0);$$
  \item $V$ is said to be \textbf{positive semidefinite} if $$V(x,y) \geq 0\text{ for all }(x,y) \neq (0,0);$$
  \item $V$ is said to be \textbf{negative definite} if $$V(x,y) < 0\text{ for all }(x,y) \neq (0,0);$$
  \item $V$ is said to be \textbf{negative semidefinite} if $$V(x,y) \leq 0\text{ for all }(x,y) \neq (0,0);$$
  \end{enumerate}

  $V$ is said to provide a zeroth order test if it is positive definite. In this situation, we can conclude that $(0,0)$ is stable.

  If $V$ is positive \emph{semidefinite}, we get a first order test. To see what this means, let $(x(t),y(t))$ be a solution curve. We then have
  \begin{equation}
    \frac{d}{dt} V(x(t),y(t))
    = V_x\frac{dx}{dt} + V_y\frac{dy}{dt}
    = V_xF(x,y) + V_yG(x,y).
  \end{equation}
  This calculation motives the definition
  \begin{equation}
    \dot V(x,y) := V_x(x,y)F(x,y) + V_y(x,y)G(x,y).
  \end{equation}
  We have a pair of functions $(V,\dot V)$. $V$ tells us if we have a stable critical point, and the sign of $\dot V$ (the first order test) tells us if it is a source or a sink. This is made precise by the following theorem.

  \medskip

\item \underline{Stability Theorem.} Suppose $(0,0)$ is an isolated critical point, an furthermore that the following conditions hold:
  \begin{enumerate}
  \item $V(0,0) = 0$;
  \item $V$ is positive definite;
  \item $\dot V$ is negative semidefinite.
  \end{enumerate}
  Then $(0,0)$ is stable. Moreover if $\dot V$ is negative definite, we can conclude that $(0,0)$ is asymptotically stable.

  Notice that $\dot V(0,0) = 0$ follows from the positive definiteness of $V$ (which means $(0,0)$ is a local minimum of $V$).

  \underline{Example (mass-spring system, continued):}
  \begin{equation}
    \begin{cases}
      (0,0) = \text{critical point},\\
      V(x,y) = \frac{1}{2}my^2 + \frac{1}{2}kx^2.
    \end{cases}
  \end{equation}
  Then
  \begin{equation}
    \dot V(x,y) = V_x\dot x + V_y\dot y
    = kxy + my\left(-\frac{k}{m}x - \frac{c}{m}y\right)
    = -cy^2 \leq 0,
  \end{equation}
  so $\dot V$ negative semidefinite. However, it is not negative definite (because there are nonzero $(x,y)$ with $y=0$).
  By the theorem, $(0,0)$ is stable, but we cannot say whether or not it is asymptotically stable.

  \medskip

\item \underline{Question:} What is $V$, in general?

  This is the trouble of applying Lyapunov's method; we need to find an appropriate $V$ for our system. There are no formulas to tell us what choice to make. Here is a function you can try, that usually works for simple systems: $$V(x,y) = ax^2 + bxy + cy^2\quad\quad \text{ (positive definite if $a>0$ and $4ac-b^2>0$)}.$$

  \underline{Example}
  \begin{equation}
    \begin{cases}
      \dot x = -x-2y\\
      \dot y = 2x-y^3
    \end{cases}
  \end{equation}
  We try the second order homogeneous quadratic form of $V$, to analyse the critical point $(0,0)$. Keeping the coefficients of $V$ generic, we compute
  \begin{equation}
    \begin{split}
      \dot V &= \frac{\partial V}{\partial x}\dot x + \frac{\partial V}{\partial y}\dot y\\
      &= (2ax + by)(-x-2y) + (bx + 2cy)(2x-y^3)\\
      &= (-2a + b)x^2 + (-4a - 2b + 4c)xy - bxy^3 - 2cy^4.
    \end{split}
  \end{equation}
  We want to make this negative (semi-)definite. Looking at this generic $\dot V$, we recognise that the $x^2$ and and $y^4$ terms are ``safe'' so long as $c \geq 0$ and $-2a + b \leq 0$, and ideally we would try to get these terms strictly negative. We can kill the ``problematic'' $xy$ and $xy^3$ terms if we set $b = 0$ and $a = c$.\footnote{These terms are problematic because $xy$ and $y^3$ are themselves not positive or negative definite. Instead they have a saddle/inflection point at the origin, so they make it hard for $\dot V$ to be nonpositive everywhere.}

  Moreover, if we make $a = c$ positive, we get the most we could ask for, because now $x^2$ and $y^4$ have the right sign and force $\dot V$ to be negative definite:
  \begin{equation}
    \dot V \propto -x^2 - y^4.
  \end{equation}
  If we choose $a = c = 1$, then
  \begin{equation}
    V(x,y) = x^2 + y^2
  \end{equation}
  is positive definite and
  \begin{equation}
    \dot V(x,y) = -2x^2 -2y^2
  \end{equation}
  is negative definite, and we can conclude by the Stability Theoem that $(0,0)$ is an asymptotically stable saddle point.

  \medskip

\item \underline{Instability Theorem.} Suppose $(0,0)$ is an isolated critical point and the following conditions hold:
  \begin{enumerate}
  \item $V(0,0) = 0$;
  \item $V$ is weakly positive definite: there exists a sequence $(x_n,y_n)$ converging to zero such that $V(x_n),y_n) > 0$ for all $n$;
  \item $\dot V$ is positive definite.
  \end{enumerate}
  Then $(0,0)$ is unstable.

  The instability theorem is clearly complementary to the stability theorem, and covers some (but not all) of the cases where the stability theorem fails. Thus the stability/instability theorems give a partial characterisation of the stability of critical points.

  \underline{Example:} Consider the system
  \begin{equation}
    \begin{cases}
      \dot x = x\\
      \dot y = -y
    \end{cases}
  \end{equation}
  with the Lyapunov function
  \begin{equation}
    V(x,y) = x^2 - y^2.
  \end{equation}
  Now, $V(x,y)$ is not positive definite, and cannot be coerced into satisfying the conditions of the stability theorem. However, $V$ is weakly positive definite, since it is positive on $(x_n,y_n) = (1/n,0)$. We compute
  \begin{equation}
    \dot V = (2x)(x) + (-2y)(-y)
    = 2(x^2+y^2),
  \end{equation}
  so $\dot V$ is positive definite. By the instability theorem, we conclude that the origin is an unstable critical point.

  Notice how $V$ was chosen. It is inspired by the parabolic $V$ we used for the mass-spring system, but with the sign of the $y^2$ flipped. This choice was made so that the signs of its partial derivatives cancel out the signs of $\dot x$ and $\dot y$.

\end{enumerate}

\medskip

\subsection{Periodic solutions and limit cycles}

\begin{enumerate}
\item \underline{Motivation} Consider, as an example, the harmonic oscillator from classical mechanics. This system is described by the equation of motion
  \begin{equation}
    \ddot x = \frac{k}{m}x = 0,
  \end{equation}
  and is the special case of the mass-spring system with $c = 0$. Absorbing the constants $k$ and $m$ into a single constant $\omega_0^2 = \sqrt{k}{m}$, we have
  \begin{equation}
    \ddot x = \omega_0^2x = 0.
  \end{equation}
  Going back to our study of second order systems, we recognize the characteristic polynomial
  \begin{equation}
    p(z) = z^2 + \omega_0^2 \quad\implies\quad
    \text{roots are }\pm\omega_0 i,
  \end{equation}
  so the general solution is
  \begin{equation}
    x(t) = c_1\sin(\omega_0 t) + c_2\cos(\omega_0 t).
  \end{equation}

  Looking at the phase space plot of the solution curves, we see that the origin is a centre.

  But alternatively, we can reduce the equation to a first order system
  \begin{equation}
    \begin{cases}
      \dot y = -\omega_0^2x\\
      \dot x = y
    \end{cases},
  \end{equation}
  from which the analysis developed in the preceding sections produces the same result.

  It is a result from physics that almost everything is very nearly a harmonic oscillator most of the time. In math, this is kind of idea is made formal by our linear approximation results for locally linear systems. But we want to go beyond talking about stability of equilibrium solutions; we want to capture the notion of a stable periodic solution (which will in simple examples look like the solution curves of the harmonic oscillator). As an extension of the notion of asymptotic stability, we will want to study systems where the solution curves ``approach'' the kind of elliptic or periodic nature of the harmonic oscillator trajectory.

  \medskip

\item \underline{General Setting}

  We will have a first order system
  \begin{equation}
    \begin{cases}
      \dot x = F(x,y)\\
      \dot y = G(x,y)
    \end{cases}
  \end{equation}

  We make the following definitions.

  \underline{Definitions.} Let $\vec x(t) = (x(t),y(t))$. If $\vec x(T + t) = \vec x(t)$ for some $T$ and for all $t\in \RR$, then $\vec x$ is called a \textbf{periodic solution}. The smallest such positive real number $T$ is called the \textbf{period} of the solution. Notice that the solution curves of periodic solutions are all closed curves (and include, for example, all solutions to the harmonic oscillator).

  \medskip

\item \underline{Example (a linear system).}
  \begin{equation}
    \begin{cases}
      \dot x = 2x-y\\
      \dot y = x + 2y
    \end{cases}
    \quad\quad
    \begin{pmatrix}
      \dot x \\ \dot y
    \end{pmatrix}
    =
    \begin{pmatrix}
      2 & -1\\
      1 & 2
    \end{pmatrix}
    \begin{pmatrix}
      x \\ y
    \end{pmatrix}.
  \end{equation}
  The characteristic polynomial of the matrix is $\lambda^2 - 4\lambda + 5$, so the eigenvalues are $\lambda = 2\pm2i$. Notice:
  \begin{enumerate}
  \item The positivity of the real part of $\lambda$ causes the solutions to ``blow up'': $(0,0)$ is an unstable equilibrium solution;
  \item The nonzero imaginary part causes the solutions to circulate around the origin, so we find that the origin is a spiral source.
  \end{enumerate}

  \medskip

\item \underline{Example (a locally linear system).}

  We consider a locally linear system that reduces to the above system in the linear approximation:
  \begin{equation}
    \begin{cases}
      \dot x = 2x-y - x(x^2 + y^2)\\
      \dot y = x+2y - y(x^2 + y^2)
    \end{cases}
  \end{equation}
  We have already solved the linear behaviour of this system, so let's analyse the global behaviour. The $x^2 + y^2$ showing up everywhere on the RHS motivates switching to polar coordinates.

  1. \underline{Radial dynamics}\\
  We first derive and study the equation of motion for $r$. Plugging $r^2 = x^2 + y^2$ into the above system, and considering its derivative $\ds r\frac{dr}{dt} = x\frac{dx}{dt} + y\frac{dy}{dt}$, we have
  \begin{equation}
    \begin{split}
      r\frac{dr}{dt} &= x(2x-y - x(x^2+y^2)) + y(x+2y-y(x^2+y^2))\\
      &= 2x^2 + 2y^2 - (x^2+y^2)(x^2+y^2)\\
      &= 2r^2 - r^2r^2\\
      &= r^2(2-r^2),
    \end{split}
  \end{equation}
  which gives the equation of motion
  \begin{equation}
    \frac{dr}{dt} = r(2-r^2).
  \end{equation}

  2. \underline{Angular dynamics}\\
  Now we fix $r$, and focus on the derivative with respect to $\theta$. Switching to dot notation for time derivatives (I am tired of writing $d/dt$), and using $x = r\cos\theta,\,y = r\sin\theta$, we calculate
  \begin{equation}\label{eq:angular-dynamics1}
    \begin{split}
      x\dot y - y\dot x &= r\cos\theta\left(r\cos\theta \dot\theta + \dot r\sin\theta\right) - r\sin\theta\left(r(-\sin\theta)\dot\theta + \dot r\cos\theta\right)\\
      &= r^2\cos^2\theta\dot\theta + r^2\sin^2\theta\dot\theta\\
      &= r^2\dot\theta.
    \end{split}
  \end{equation}
  Note that if $x$ and $y$ were interpreted as the cartesian coordinates of some particle in space, $r^2\dot\theta$ is proportional to the angular momentum of the particle, a conserved quantity! (In physics, when the forces are conservative. This example certainly doesn't represent such a system. In fact, the following shows that this is only conserved in the trivial case where the trajectory is circular. Sadness.)

  Now using the right hand side of the original system, we expand the left hand side of (\ref{eq:angular-dynamics1}) to get
  \begin{equation}
    \begin{split}
      r^2\frac{d\theta}{dt} &= x\dot y - y\dot x\\
      &= x(x+2y - y(x^2+y^2)) - y(2x-y-x(x^2+y^2))\\
      &= x^2 + y^2\\
      &= r^2
    \end{split}
  \end{equation}
  which implies
  \begin{equation}
    \frac{d\theta}{dt} = 1.
  \end{equation}

  3. \underline{Global dynamics}\\
  From the equation of motion for $r$, it can be seen that $r = \sqrt2$ is a stable critical point for the one dimensional system $r = r(t)$. The full 2D system is now described in polar coordinates by
  \begin{equation}\label{eq:polar-coordinates-new-first-order-system}
    \begin{cases}
      \dot r = r(2-r^2)\\
      \dot \theta = 1
    \end{cases}
  \end{equation}
  As a remark, note that we can recover the original system by simply differentiating $x = r\cos\theta$ and $y = r\sin\theta$:
  \begin{equation}
    \begin{split}
      \frac{dx}{dt} &= r(-\sin\theta)\frac{d\theta}{dt} + \frac{dr}{dt}\cos\theta\\
      &= r(-\sin\theta)(1) + (2r - r^3)\cos\theta\\
      &= -y + 2x - x(x^2 + y^2)
    \end{split}
  \end{equation}
  (the $dy/dt$ calculation is similar). Therefore (\ref{eq:polar-coordinates-new-first-order-system}) contains exactly the same information as the original expression for the system in terms of $x$ and $y$. (If you are worried about the singularity in the polar coordinates at the origin, note that the values of $dx/dt$ and $dy/dt$ can be filled in at this point using continuity of the derivatives).

  In fact, the equation $\dot\theta = 1$ tells us that critical points can only correspond to points where $\theta$ fails to be a good coordinate; i.e. it is very clear in the polar coordinate form that the only critical point is the origin. But to go beyond critical points and to talk about limit cycles, the $\dot r$ equation is of interest. Factoring into linear polynomials, we have
  \begin{equation}
    \frac{dr}{dt} = r(\sqrt 2 - r)(-\sqrt{2} -r).
  \end{equation}
  As mentioned above, $r = \sqrt{2}$ is a stable value of $r$ in the sense that any solution $r(t),\theta(t)$ such that $r(t_0) = \sqrt{2}$ for some $t_0$ will remain at $r = \sqrt{2}$ forever (it is a circular periodic solution), and moreover we have the following analysis. Consider the interior and exterior regions of the plane defined by the circle $r = \sqrt{2}$. In the interior region, $r < \sqrt{2} \implies dr/dt > 0$ and we have a spiral solution curve converging outwards to $r = \sqrt{2}$. In the exterior region, $r > \sqrt{2} \implies dr/dt < 0$ and we have a spiral converging inwards to $r = \sqrt{2}$. This behaviour indicates that the circle at $r = \sqrt{2}$ has some serious stability property.

  Call the circle of radius $r = \sqrt{2}$ $C_{\sqrt{2}}$. We call $C_{\sqrt{2}}$ a \textbf{stable limit cycle} in this case, meaning every solution curve converges to $C_{\sqrt{2}}$. Similarly, an \textbf{unstable limit cycle} is a periodic solution with all other solution curves spiralling away. Finally, a \textbf{semistable limit cycle} is a periodic solution such that some other solutions converge to it, but not every solution. One can easily cook up examples of these other cases. I'll throw in an example of a system with all three kinds of limit cycles. Let
  \begin{equation}
    \begin{cases}
      \dot r = (r-1)(r-2)^2(r-3)\\
      \dot \theta = -1
    \end{cases}
  \end{equation}
  Clearly $r = 1,2,3$ correspond to limit cycles of the system, and because $\dot\theta < 0$ all solution curves now proceed clockwise around the origin. We split the analysis into cases, again corresponding to the various regions on the phase plot.

  \underline{$r < 1$}: $dr/dt > 0$ and so solution curves spiral out towards $r = 1$;\\
  \underline{$1 < r < 2$}: $dr/dt < 0$ and so solution curves spiral away from $r=2$ towards $r = 1$;\\
  \underline{$2 < r < 3$}: $dr/dt < 0$ and so solution curves spiral away from $r=3$ towards $r = 2$;\\
  \underline{$r > 3$}: $dr/dt > 0$ and so solution curves spiral out away from $r = 3$.

  Observe from this qualitative analysis that $r = 1$ is a stable limit cycle, $r = 2$ is semistable, and $r = 3$ is unstable.

  \medskip

\item \underline{Theorem.} (Poincar\'e-Bendixson theorem.)

  Suppose $R$ is a compact region (a closed and bounded subset of $\RR^2$) containing no critical points of a first order system. If $x(t)$ is a solution such that $x$ is eventually in $R$ (there exists $t_0 \in\RR$ such that $x(t) \in R$ for all $t \geq t_0$), then either\\
  1. $x(t)$ converges to a limit cycle; or\\
  2. $x(t)$ is itself a limit cycle.

  \underline{Remark:} If one can find such a region and a solution curve eventually in that region, one can conclude that a limit cycle exists without necessarily having to construct the limit cycle in the first place. In many cases, however, having an explicit solution usually means one is already capable of seeing what the limit cycle should be. Thus the theorem seems to be the most powerful when one cannot directly solve the system but is able to argue that a solution exists and is confined to such a region.

  For example, let us forget for the moment that we already know the limit cycle solution to the system analysed previously:
  \begin{equation}\label{eq:poincare-bendixson-example}
    \begin{cases}
      \dot r = r(2-r^2)\\
      \dot \theta= 1
    \end{cases}
  \end{equation}
  Let $R$ be the closed annulus $R = \left\{(r,\theta)\big|1 \leq r \leq 2 \right\}$. Then $R$ is closed and bounded and does not contain the only critical point of the system, namely the origin. All that remains is to choose a solution curve eventually in $R$. But if we let $r(t_0) = 1$, $\theta(t_0) = 0$, the resulting curve starts out in $R$ and satisfies $dr/dt > 0$ whenever $r < \sqrt{2}$ and $dr/dt < 0$ whenever $r > \sqrt{2}$. Since such curves can only pass through the boundaries $r = 1$ and $r=2$ of $R$ on their way into the region, we conclude that $(r(t),\theta(t))$ never leaves $R$. By the Poincar\'e-Bendixson theorem, there exists a limit cycle of (\ref{eq:poincare-bendixson-example}) and moreover it lies in the annulus $1 \leq r \leq 2$.

\end{enumerate}

\subsection{Some applications}

\begin{enumerate}

\item Logistic equation (a biological application).

  Let $x$ denote the population size of some species. A prototypical biological model is the \textbf{logistic equation}, which reads
  \begin{equation}\label{eq:logistic-equation}
    \frac{dx}{dt} = rx\left(1-\frac{x}{k}\right).
  \end{equation}
  When $x \ll k$, the logistic equation reduces approximately to the linear equation for exponential growth $dx/dt \approx rx$. $k$ is called the \textbf{carrying capacity}, because $dx/dt = 0$ when $x = k$, so $k$ is a critical point (in fact a stable critical point) of the first order system where the population is unchanging. We will look at several predator prey systems modelled on the logistic equation.

\item \underline{Example}
  \begin{equation}
    \begin{cases}
      \frac{dx}{dt} = x(1 - x - y)\\
      \frac{dy}{dt} = y\left(\frac{3}{4} - y - \frac{1}{2}x\right)
    \end{cases}
  \end{equation}
  This system secretly contains two logistic equations (the $x(1-x)$ and $y(3/4-y)$ parts of the right hand side), as well as an ``interaction'' between the two species describing their predator/prey relationship. We can apply our previous methods to study this nonlinear first order system. First of all, the critical points are the intersection points of the sets where $dx/dt = 0$ and $dy/dt = 0$. Looking at the form of the right hand side, we can see that both of these sets are just unions of lines. This motivates the following terminology.

  The \textbf{$x$-nullcline} is the union of the lines $x = 0$ and $1 - x - y = 0$ (the line passing through $(1, 0)$ and $(0, 1)$).

  The \textbf{$y$-nullcline} is the union of the lines $y = 0$ and $\frac{3}{4} - y - \frac{1}{2}x = 0$ (the line passing through $\left(\frac{3}{2}, 0\right)$ and $\left(0, \frac{3}{4}\right)$).

  In general, the $x$- (respectively, $y$-)nullcline is the set of points on which $dx/dt$ (respectively, $dy/dt$) vanishes.

  The critical points are the intersection points of the $x$-nullcline and the $y$-nullcline. They are $(0, 0)$, which represents both species being extinct; $(1,0)$, with species $x$ at carrying capacity and $y$ extinct; $\left(0,\frac{3}{4}\right)$, with species $x$ extinct and $y$ at carrying capacity; and $\left(\frac{1}{2},\frac{1}{2}\right)$, which is the intersection point of the two nontrivial nullclines.

  Drawing the nullclines on a phase diagram splits the domain (in this case the first quadrant of the $xy$-plane) into regions which we can label by the signs of $dx/dt$ and $dy/dt$. Looking at the resulting diagram, we can classify $(1/2,1/2)$ is a stable critical point. In biology it is called a \textbf{coexistence equilibrium}.

  \medskip

\item \underline{Example} (Lotka-Volterra equation.) Consider the Lotka-Volterra system
  \begin{equation}
    \begin{cases}
      \frac{dx}{dt} = x(2-y) = F(x,y)\\
      \frac{dy}{dt} = y(-1 + x) = G(x,y)
    \end{cases}
  \end{equation}
  A comment on the coefficients. The ``2'' in $dx/dt$ represents a growth rate of the prey $x$ in this model, and the ``-1'' in $dy/dt$ represents a decay rate of the predator $y$.

  We can do a locally linear system analysis on this equation. We should compute the Jacobian
  \begin{equation}
    J(x,y) =
    \begin{pmatrix}
      F_x & F_y\\
      G_x & G_y
    \end{pmatrix}
    =
    \begin{pmatrix}
      2-y & -x\\
      y & -1+x
    \end{pmatrix}.
  \end{equation}
  Let's look at the critical point $(x,y) = (1,2)$. The linear behaviour of the system about this point is given by
  \begin{equation}
    J(1,2) =
    \begin{pmatrix}
      0 & -1\\
      2 & 0
    \end{pmatrix}
  \end{equation}
  This is a nonsingular matrix, but upon computing the eigenvalues we find $\lambda = \pm \sqrt{2}i$. Unfortunately, this is precisely the case (eigenvalues purely imaginary) when locally linear analysis is inconclusive; no results from the linear system carry over to the original nonlinear one.

  Now let us consider the exact solution curves to the nonlinear system. To solve for these, we find the autonomous system
  \begin{equation}
    \frac{dy}{dx} = \frac{dy/dt}{dx/dt} = \frac{y(-1+x)}{x(2-y)} = \left(\frac{y}{2-y}\right)\left(\frac{-1+x}{x}\right)
  \end{equation}
  Luckily, this equation is fairly easy to solve, as it is separable. We rearrange and simplify to get
  \begin{equation}
    \left(\frac{2}{y} - 1\right)dy = \left(\frac{-1}{x} + 1\right)dx,
  \end{equation}
  and integrating, we have
  \begin{equation}
    2\ln y - y = -\ln x + x + C.
  \end{equation}
  The absolute values are ignored because $x$ and $y$ represent populations in this model; we are not considering negative $x$ and $y$. We can capture these solution curves by writing them as the level curves of a function $H(x,y)$, by writing
  \begin{equation}
    H(x,y) = 2\ln y - y + \ln x - x.
  \end{equation}

  \medskip

\item \underline{Hamiltonian systems.} The setting is as follows.

  1. $x$ stands for the generalised position (or configuration space variable);
  2. $y$ stands for the generalised momentum.

  In physics, one might use $q$ and $p$ (or $x$ and $p$) for generalised position and momentum.

  The \textbf{Hamiltonian} is defined to be a smooth function $H(x,y)$, that represents the total energy of the system. The Hamiltonian system (Hamilton's equations of motion) are
  \begin{equation}
    \dot x = \frac{\partial H}{\partial y},\quad\quad\quad
    \dot y = -\frac{\partial H}{\partial x}
  \end{equation}
  In a sense, the first equation captures the idea that a particle with more momentum ``moves faster'', and the second equation is the corresponding generalisation of Newton's second law.

  \medskip
  \underline{Example} (Newtonian system.)

  We let $x$ denote position (along a line) and $y = m\dot x$ the classical momentum of a point particle. From physics, $H(x,y)$ takes the form
  \begin{equation}
    H(x,y) = \frac{1}{2}m\dot x^2 + V(x)
    = \frac{y^2}{2m} + V(x)
  \end{equation}
  where $V$ is a function of $x$ \emph{only}, representing the potential energy of a conservative force (all position-dependent forces are conservative in 1 dimension, but the situation becomes more complicated in higher dimensions).

  The first Hamilton equation becomes
  \begin{equation}
    \frac{\partial H}{\partial y} = \frac{2y}{2m} = \frac{y}{m} = \dot x,
  \end{equation}
  and the second Hamilton equation becomes
  \begin{equation}
    \dot y = -\frac{\partial H}{\partial x} = -\frac{\partial V}{\partial x} = F,
  \end{equation}
  where $F$ is the force corresponding to the potential $V$. This is exactly Newton's second law, which says that the rate of change of momentum is equal to the force on the particle.

  \underline{In general...}\\
  Back to the general case, we can define the phase space variable $u = [x, y]^T$, and summarise Hamilton's equations by
  \begin{equation}
    \dot u = f(u) =
    \begin{pmatrix}
      0 & 1\\
      -1 & 0
    \end{pmatrix}
    \begin{pmatrix}
      \partial H/\partial x\\
      \partial H/\partial y
    \end{pmatrix}(u)
  \end{equation}
  (the vector on the right is a function of $u$).

  From this equation we can derive a very important property of Hamiltonian systems, which is Liouville's theorem, or the divergence-free property. To derive this, we simply evaluate
  \begin{equation}
    \nabla\cdot f = \frac{\partial f_1}{\partial x} + \frac{\partial f_2}{\partial y}
    = \frac{\partial}{\partial x}\left(\frac{\partial H}{\partial y}\right) + \frac{\partial}{\partial y}\left(\frac{\partial H}{\partial x}\right)
    = \frac{\partial^2H}{\partial x \partial y} - \frac{\partial^2H}{\partial y\partial x}.
  \end{equation}
  By Clairout's theorem, under very mild regularity assumptions (continuity of the second derivatives) we can exchange the order of differentiation and in this case we can conclude that $\nabla\cdot f = 0$. A very physically interesting consequence of this divergence-free property, which we will not go into here, is that areas in phase space ($(x,y)$-space) are preserved by evolution under Hamilton's equations of motion. When interpreted in many body mechanics, this becomes a continuity equation on the phase space that describes conservation of probability in some sense.

  Now we do some locally linear analysis. The Jacobian is
  \begin{equation}
    J =
    \begin{pmatrix}
      \partial f_1/\partial x & \partial f_1/\partial y\\
      \partial f_2/\partial x & \partial f_2/\partial y
    \end{pmatrix}
    =
    \begin{pmatrix}
      \partial^2H/\partial x\partial y & \partial^2H/\partial y^2\\
      -\partial^2H/\partial x^2 & -\partial^2H/\partial y\partial x
    \end{pmatrix}
  \end{equation}
  Let's look at the trace of $J$, the sum of the diagonal entries. From our previous calculation, and looking at the above matrix, we can see that $\tr(J) = 0$. On the other hand, the trace of $J$ is the sum of its eigenvalues $\tr(J) = \lambda_1 + \lambda_2$, so this means we cannot have any sources or sinks, spiral or otherwise. Why? Because a source requires the real part of both eigenvalues positive, and a sink require the real part of both eigenvalues to be negative. In either case, we would fail to satisfy $\lambda_1 + \lambda_2 = 0$. So the Hamiltonian system has \underline{no sources or sinks} by locally linear analysis.

  \underline{Lyapunov analysis} We can view $H(x,y)$ as a Lyapunov function on the Hamiltonian system. We have
  \begin{equation}
    \begin{split}
      \frac{d}{dt}H(x(t),y(t)) &= \frac{\partial H}{\partial x}\frac{\partial x}{\partial t} + \frac{\partial H}{\partial y}\frac{\partial y}{\partial t}\\
      &= \frac{\partial H}{\partial x}\frac{\partial H}{\partial y} - \frac{\partial H}{\partial y}\frac{\partial H}{\partial x}\\
      &= 0.
    \end{split}
  \end{equation}
  This result is called \textbf{conservation of energy}, in analogy with the physical interpretation. The consequence is that every solution curve is a level curve of $H$ (since $H(x,y)$ is equal to a constant).

  \medskip
  \underline{Another example.}
  \begin{equation}\label{eq:another-hamiltonian-system}
    \begin{cases}
      \dot x = 8y + ax^2y\\
      \dot y = -2x + 2xy^2
    \end{cases}
  \end{equation}
  We will: (a) find a value of $a$ such that the system is Hamiltonian, and construct the associated Hamiltonian function; (b) analyse the critical points using the Jacobian. We want
  \begin{align}
    \frac{\partial H}{\partial y} &= 8y + ax^2y\\
    -\frac{\partial H}{\partial x} &= -2x+2xy^2.
  \end{align}
  Notice that the derivative of the first expression with respect to $x$ must be equal to (negative) the derivative of the second expression with respect to $y$ (exchanging the partial derivatives by Clairout's theorem). In other words,
  \begin{align}
    \frac{\partial}{\partial x}\left(8y + ax^2y\right) &= \frac{\partial}{\partial y}\left(2x-2xy^2\right)\quad\forall (x,y)\\
    \iff 2axy &= -4xy\quad\forall(x,y)\\
    \iff a &= -2.
  \end{align}
  If $a$ is chosen to be $-2$, the system is therefore Hamiltonian and all we have to do is construct $H$.
  \begin{equation}
    \begin{split}
      H(x,y) &= \int\frac{\partial H}{\partial x}dx\\
      &= \int(2x-2xy^2)dx\\
      &= x^2 - x^2y^2 + C(y),
    \end{split}
  \end{equation}
  where $C(y)$ is an as-of-yet undetermined function of $y$. Carrying out the integration with respect to $y$, we have
  \begin{equation}
    \begin{split}
      H(x,y) &= \int\frac{\partial H}{\partial y}dy\\
      &= \int(8y - 2x^2y)dy\\
      &= 4y^2 - x^2y^2 + D(x).
    \end{split}
  \end{equation}
  Comparing the two forms of $H(x,y)$, we deduce that $D(x) = x^2$ plus a constant (or equivalently $C(y) = 4y^2$ plus a constant). Thus
  \begin{equation}
    H(x,y) = x^2 - x^2y^2 + 4y^2 + C
  \end{equation}
  where $C$ is an arbitrary constant.

  Now we can look for critical points. Setting the right hand side of (\ref{eq:another-hamiltonian-system}) to zero, and plugging in $a = -2$, we have the possibilities
  \begin{equation}
    \begin{cases}
      x^2 = 4,\quad y = 0\\
      x = 0,\quad y^2 = 1
    \end{cases}
  \end{equation}
  The critical points are thus $(0,0)$ and $(\pm 2,\pm 1)$. By taking all the partial derivatives, one finds the Jacobian
  \begin{equation}
    J(x,y) =
    \begin{pmatrix}
      -4xy & 8 - 2x^2\\
      -2+2y^2 & -4xy
    \end{pmatrix}
  \end{equation}
  For $J(0,0)$, we find the characteristic equation $\lambda^2 + 16$. But this polynomial only has purely imaginary roots $\pm 4i$, and this is the case where locally linear analysis fails. So the test is inconclusive for $(0,0)$.\\
  For $J(2,1)$, win find the characteristic equation $(\lambda-8)(\lambda+8)$, so $J(2,1)$ has real eigenvalues of opposite sign. So $(2,1)$ is an unstable saddle point.

  Notice that the trace of each of the Jacobians vanishes! This is guaranteed by our general analysis of Hamiltonian systems. So a good check of whether you did your arithmetic correctly for Hamiltonian systems is to eyeball the trace of $J$ and make sure it is zero.

\end{enumerate}

\section{Bifurcations in First Order Differential Equations}

The reference for this section is Chapter 6 of the Paul C. Matthews textbook.

We begin this section with an example. Consider the differential equation
\begin{equation}
  \frac{dx}{dt} = f(x,\mu)\\
  = 25x^2 + \mu^2 - 100
\end{equation}
where $\mu$ is a real parameter. Consider the critical points ($dx/dt = 0$). They will be functions of $\mu$. Setting the right hand side to zero,
\begin{equation}\label{eq:first-example-of-branches}
  25x^2 + \mu^2 = 100
  \implies \frac{x^2}{4} + \frac{\mu^2}{100} = 1
  \implies \left(\frac{x}{2}\right)^2 + \left(\frac{\mu}{10}\right)^2 = 1.
\end{equation}
This is the equation for an ellipse with $x$ between $-2$ and $2$, and $\mu$ between $-10$ and $10$. Drawing the ellipse on an $x\mu$-plane, we see that (\ref{eq:first-example-of-branches}) does not do to define $x$ (the critical point) as a function of $\mu$. For there are two possible solutions! We can however capture both critical points with two functions $x(\mu)$:
\begin{equation}
  x(\mu) = +\frac{1}{5}\sqrt{100 - \mu^2}\quad\quad -10 < \mu < 10
\end{equation}
and
\begin{equation}
  x(\mu) = -\frac{1}{5}\sqrt{100 - \mu^2}\quad\quad -10 < \mu < 10.
\end{equation}
These two functions are called \textbf{branches} (the top branch and the bottom branch, espectively). Notice we have left the ``bifurcation points'' $\mu = \pm10$ out of the definition of the branches. This is intentional, and we will return to these points later.

We can look at the stability of the two critical points by using the derivative $f_x = 50x$. In the top branch, $f_x > 0$. We claim that for the upper branch,
\begin{equation}\label{eq:top-branch-stability}
  \begin{cases}
    \dot x = f(x,\mu) > 0,\quad x > x(\mu)\\
    \dot x = f(x,\mu) < 0,\quad x < x(\mu)
  \end{cases}
\end{equation}
and hence $x(\mu)$ is unstable. To see why this is true, write
\begin{equation}
  f_x(x(\mu),\mu) = \lim_{x\to x(\mu)}\frac{f(x,\mu)-f(x(\mu),\mu)}{x-x(\mu)}
  \approx \frac{f(x,\mu)-f(x(\mu),\mu)}{x-x(\mu)} = \frac{f(x,\mu)}{x-x(\mu)}\quad
  \text{when $x\approx x(\mu)$}.
\end{equation}
Thus for $x>x(\mu)$, $f_x$ and $f(x,\mu)$ have the same sign, and when $x < x(\mu)$, $f_x$ and $f(x,\mu)$ have opposite sign. Since $f_x = 50x > 0$ near the top branch, we have (\ref{eq:top-branch-stability}).

A similar analysis for the bottom branch shows that all those critical points $x=x(\mu)$ are all stable. In general, for the first order system
\begin{equation}
  \frac{dx}{dt} = f(x,\mu)
\end{equation}
with parameter $\mu$, all the critical points at which $f_x > 0$ are unstable, and those for which $f_x < 0$ are stable.

Notice that the obstruction to continuing our analysis past $\mu = 10$ (the bifurcation point) is a change in the number of critical points (from two to one to zero).

\begin{enumerate}
\item \underline{Generalised form and some definitions.}

  We consider the first order system
  \begin{equation}\label{eq:bifurcations-system}
    \frac{dx}{dt} = f(x,\mu),\quad\quad x\in \RR,\quad\mu\in\RR.
  \end{equation}
  The \textbf{critical points} are the $x^*$ such that $f(x^*,\mu) = 0$ (so the critical points depend on the parameter $\mu$).

  There are three stability cases for critical points. To summarise, we have said that
  \begin{enumerate}
  \item $f_x(x^*,\mu)>0$: $x^*$ is unstable;
  \item $f_x(x^*,\mu)<0$: $x^*$ is stable;
  \end{enumerate}
  but we haven't talked about the case $f_x(x^*,\mu)=0$. Before discussing this case, we will talk about bifurcations.

  \underline{Bifurcations.} An informal definition (we will give a more formal one later), is the following. We say \emph{a bifurcation occurs at $(x^*_c,\mu_c)$ if there is a qualitative change in the phase line of $\dot x = f(x,\mu)$ as $\mu$ passes through $\mu_c$.} In this situation, $(x_c,\mu_c)$ is called a \textbf{bifurcation point}.

  We will also need an informal statement of the \underline{Implicit Function Theorem.} Assume $f(x_0,\mu_0) = 0$, and $f_x(x_0,\mu_0) \neq 0$. (We have in mind that $x_0$ is a critical point for $\mu = \mu_0$, but the statement we are about to make is very general and the starting point for manifold theory, incidentally). Then near $(x_0,\mu_0)$, the equation
  \begin{equation}
    f(x,\mu) = 0
  \end{equation}
  admits a unique function $x = x(\mu)$ with $x(\mu_0) = x_0$ such that
  \begin{equation}
    f(x(\mu),\mu) = 0.
  \end{equation}
  Moreover, its derivative is given by
  \begin{equation}\label{eq:implicit-fn-derivative}
    \frac{dx(\mu)}{d\mu} = -\frac{f_\mu(x(\mu),\mu)}{f_x(x(\mu),\mu)} = -\frac{f_\mu}{f_x}\bigg|_{(x(\mu),mu)}.
  \end{equation}

  There are several ideas captured by this theorem. The first is that since the derivative $f_x$ is nonzero at $(x_0,\mu_0)$, the function $f(x,\mu)$ is approximated by a linear function
  \begin{equation}\label{eq:linear-approx-implicit-fn}
    f(x,\mu)\simeq f(x_0,\mu_0) + f_x(x_0,\mu_0)(x-x_0) + f_\mu(x_0,\mu_0)(\mu-\mu_0)
    \implies f_x(x_0,\mu_0)(x-x_0) \simeq -f_\mu(x_0,\mu_0)(\mu-\mu_0)
  \end{equation}
  with the coefficient in front of $(x-x_0)$ nonzero. And we know that the linear function of $x$ on the right hand side can be ``inverted'' in this case. The implicit function theorem is really a statement about when a differentiable function $\mu(x)$ can be inverted (locally) to give a differentiable function $x(\mu)$. It says it is enough that the first derivative is nonzero at that point.

  If you take the limit $\mu-\mu_0\to 0$ in (\ref{eq:linear-approx-implicit-fn}), and rearrange the coefficients $f_x$ and $f_\mu$, you can recover (\ref{eq:implicit-fn-derivative}).

  \medskip

  \underline{Formal definition of bifurcations.} Now we can give a formal definition motivated by the implicit function theorem. Formally, a bifurcation occurs at $(x_c,\mu_c)$ if
  \begin{enumerate}
  \item $f(x_c,\mu_c) = 0$ (\emph{i.e.}, $x_c$ is a critical point of (\ref{eq:bifurcations-system}) for $\mu = \mu_c$);
  \item $f_x(x_c,\mu_c) = 0$ (\emph{i.e.} the implicit function theorem does not apply);
  \item Along the level curve $f(x,\mu) = 0$ through $(x_c,\mu_c)$, $f_x$ experiences a sign change.
  \end{enumerate}
  To see why condition (c) is important, consider the example

  \underline{Example}
  \begin{equation}
    \frac{dx}{dt} = \mu - x^3 \implies x(\mu) = \mu^{1/3},\quad\forall \mu \in \RR
  \end{equation}
  There are no bifurcations. However, considering just the first two conditions, $f(0,0) = 0$ and $f_x(0,0) = -3x^2|_{x = 0} = 0$ so both are satisfied by $x = \mu = 0$. This point is excluded by the third condition, because $f_x$ does not experience a sign change at $(0,0)$ (it is always nonpositive).

  \medskip

\item \underline{Bifurcations in general nonlinear systems.}

  Consider (\ref{eq:bifurcations-system}). We make the following assumption, clearly satisfied whenever $(0,0)$ is a bifurcation point.

  \underline{Assumption:} $f(0,0) = 0$, and $f_x(0,0) = 0$ (the implicit function theorem does not apply).

  In the spirit of the locally linear analysis we did earlier, we Taylor expand the original system. But this time, we do it to second order. For $|x|,|\mu| \ll 1$,
  \begin{equation}
    \frac{dx}{dt} = f(x,\mu) \simeq f(0,0) + f_x(0,0)x + f_\mu(0,0)\mu + \frac{1}{2}f_{xx}(0,0)x^2 + f_{x\mu}(0,0)x\mu + \frac{1}{2}f_{\mu\mu}(0,0)\mu^2.
  \end{equation}
  Using our assumptions to kill the first two terms, we have $dx/dt$ as the second-order polynomial
  \begin{equation}
    \frac{dx}{dt} \simeq f_\mu(0,0)\mu + \frac{1}{2}f_{xx}(0,0)x^2 + f_{x\mu}(0,0)x\mu + \frac{1}{2}f_{\mu\mu}(0,0)\mu^2.
  \end{equation}
  We can use this form of the differential equation to classify the point $(0,0)$.

  \underline{Case 1} (Saddle-node bifurcations.) When $f_{\mu},\,f_{xx}\neq 0$ at $(0,0)$, but $f_{x\mu},\,f_{\mu\mu} = 0$ or can be ignored, we have the reduced differential equation
  \begin{equation}\label{eq:bifurcation-case-1}
    \frac{dx}{dt} \simeq f_\mu(0,0)\mu + \frac{1}{2}f_{xx}(0,0)x^2.
  \end{equation}
  An example is the first system we looked at, $f(x,\mu) = 25x^2 + \mu^2 - 100$, when we allow ourselves to ignore the $f_{\mu\mu}$ term. This form gives us a geometric relation between $x$ and $\mu$, using our assumptions. At critical points, we have
  \begin{equation}
    x^2 \simeq -\frac{2f_\mu}{f_{xx}}\mu \quad\quad (x^2\sim \mu)
  \end{equation}
  (Henceforth all partial derivatives of $f$ will be assumed to be evaluated at $(0,0)$). We can make a further simplification by introducing new variables $\hat x,\,\hat\mu$, defined by
  \begin{equation}\label{eq:coord-transformation-case-one}
    x = A\hat x,\quad\quad
    \mu = B\hat\mu,\quad\quad
    A,B\in\RR
  \end{equation}
  where $A$ and $B$ are some constants to be chosen later. Then (\ref{eq:bifurcation-case-1}) becomes
  \begin{align}
    A\frac{d\hat x}{dt} = f_\mu B\hat\mu + \frac{1}{2}f_{xx}A^2\hat x^2\\
    \implies \frac{d\hat x}{dt} = f_\mu \frac{B}{A}\hat\mu + \frac{1}{2}f_{xx}A\hat x^2.
  \end{align}
  Now we exercise our freedom to choose $A,B$, and make the choice that results in
  \begin{equation}
    \frac{f_\mu B}{A} = 1,\quad\quad \frac{1}{2}f_{xx}A = -1.
  \end{equation}
  The result is that for our stability analysis, we can without loss of generality consider the system
  \begin{equation}
    \frac{dx}{dt} = \mu - x^2,
  \end{equation}
  called the \textbf{normal form} of the original equation. (For if necessary, a coordinate scaling transformation of the form (\ref{eq:coord-transformation-case-one}) could be applied to any system in Case 1.)

  For any fixed $\mu$, the set $\{x|\mu = x^2\}$ is the set of (zero, one, or two) critical points. Plotting $\mu = x^2$ in the $x\mu$-plane, one identifies several stabilty regions. (Recall the sign of $f_x$ determines stabilty by part 1.) In the simplified form, $f(x,\mu) = \mu - x^2\implies f_x = -2x$. One finds that as $\mu$ goes from negative to positive, the number of critical points goes from zero, to one at $\mu = 0$, to two critical points of opposite stability. Explicitly, for $\mu > 0$, $x = \pm\sqrt{\mu}$ are the critical points, and you can see from $f_x = -2x$ that the positive root is stable while the negative root is unstable. While the $f_x$ test fails for $\mu = 0$, going back to the definition of stability of critical points (\S\ref{sec:critical-points-general-setup}) one sees that the one critical point $x = 0$ is unstable.

  \medskip

  \underline{Case 2(a)} (Transcritical bifurcations, simplified case.) This is the case $f(0,\mu) = 0$ for all $\mu \in \RR$, where we additionally assume that $f_{xx} \neq 0$ and that $f_{\mu\mu}$ is negligible.

  For example, the logistic equation
  \begin{equation}
    \frac{dx}{dt} = x(\mu - x) = \mu x - x^2
  \end{equation}
  is of this type. Another example is the differential equation
  \begin{equation}
    \frac{dx}{dt} = x(\mu - x + e^{-x}).
  \end{equation}

  Doing the Taylor expansion again, under our bifurcation point assumption and the Case 2 condition we obtain the reduced equation
  \begin{equation}\label{eq:bifurcation-case-2}
    \frac{dx}{dt} \simeq \frac{1}{2}f_{xx}(0,0)x^2 + f_{x\mu}(0,0)x\mu
  \end{equation}
  By an appropriate coordinate transformation, the normal form should be
  \begin{equation}
    \frac{dx}{dt} = \mu x - x^2 = x(\mu - x)
  \end{equation}
  (as can be obtained by staring at (\ref{eq:bifurcation-case-2}) and choosing a sign convention). In the normal form,
  \begin{equation}
    f(x,\mu) = \mu x - x^2\quad\text{ and }\quad f_x(x,\mu) = \mu - 2x.
  \end{equation}
  For each value of $\mu$, the points $x = 0$ and $x = \mu$ are the critical points. Stability analysis via the sign of $f_x$ is left as an exercise to the reader. You should find that $x = 0$ is stable when $\mu < 0$ and unstable when $\mu > 0$; similarly $x = \mu$ is unstable for $\mu < 0$ and stable for $\mu > 0$.

  This situation is rather distinct from that of saddle-node bifurcations (Case 1). In this case, the number of critical points goes from 2 to 1 to 2, and the stability switches from one branch to the other. For completeness, the critical point $x = 0$ at $\mu = 0$ is an unstable saddle point. But one should have in mind that for bifurcation analysis it is the change in behaviour around the bifurcation point that is important, rather than the stability at the bifurcation point.

  \underline{Case 2(b)} (Transcritical bifurcation, general case.) We drop the assumption $f(0,\mu) = 0$, but retain $f_{xx} \neq 0$ and $f_{x\mu}^2 > f_{xx}f_{\mu\mu}$. (Notice that the simplified case is a special case.)

  We impose the condition $f_{x\mu}^2 > f_{xx}f_{\mu\mu}$ so that the relationship we just analysed ($x \simeq 0$ or $x\sim \mu$) still appears. To see this, accept (or prove) that under the general assumption made here, the reduced Taylor expansion of the DE is
  \begin{equation}
    \frac{dx}{dt} \simeq \frac{1}{2}f_{xx}(0,0)x^2 + f_{x\mu}(0,0)x\mu + \frac{1}{2}f_{\mu\mu}(0,0)\mu^2.
  \end{equation}
  Then to find the critical points, we set the right hand side equal to zero and solve the quadratic equation for $x$:
  \begin{equation}
    x = \frac{-f_{x\mu} \pm \sqrt{f_{x\mu}^2 - f_{xx}f_{\mu\mu}}}{f_{xx}}\mu.
  \end{equation}
  Now we can see why the constraint $f_{x\mu}^2 > f_{xx}f_{\mu\mu}$ was chosen; it ensures that two distinct critical points $x^+$ and $x^-$ exist whenever $\mu \neq 0$ (and again when $\mu = 0$, $x = 0$ is the unique critical point). In particular when $f_{x\mu}^2 \gg f_{xx}f_{\mu\mu}$, we recover the behaviour
  \begin{equation}
    x^+ \simeq 0\quad\text{ and }\quad
    x^- \simeq -\frac{2f_{x\mu}}{f_{xx}}\mu \sim \mu
  \end{equation}
  from Case 2(a). Similar stability results hold in the general case.

  \medskip

  \underline{Case 3} (Pitchfork bifurcation.) In this case the reduced Taylor approximation is
  \begin{equation}
    \frac{dx}{dt} = f_{x\mu}(0,0)x\mu + \frac{1}{3!}f_{xxx}(0,0)x^3.
  \end{equation}


\end{enumerate}
